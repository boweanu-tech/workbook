\MajorQuestionLesson{ベクトルの内積}
\SetRowSpace{5mm}

\TypeQuestionDouble{$|\vec{a}|,|\vec{b}|$ と2つのベクトルのなす角 $\theta$ が次のように与えられている。内積 $\vec{a}\cdot\vec{b}$ を求めなさい。}{dot_product_definition}
\QQRow{$|\vec{a}|=3,\ |\vec{b}|=4,\ \theta=60^\circ$}{$6$}{$|\vec{a}|=5,\ |\vec{b}|=2,\ \theta=120^\circ$}{$-5$}
\QQRow{$|\vec{a}|=2\sqrt3,\ |\vec{b}|=3,\ \theta=30^\circ$}{$9$}{$|\vec{a}|=6,\ |\vec{b}|=1,\ \theta=135^\circ$}{$-3\sqrt2$}
\QQRow{$|\vec{a}|=4,\ |\vec{b}|=7,\ \theta=90^\circ$}{$0$}{$|\vec{a}|=\sqrt2,\ |\vec{b}|=5,\ \theta=45^\circ$}{$5$}

\TypeQuestionDouble{長方形 $\mathrm{ABCD}$ において、$\mathrm{AB}=6$、$\mathrm{AD}=2\sqrt3$ であり、対角線 $\mathrm{AC}$ と辺 $\mathrm{AB}$ のなす角は $30^\circ$ である。次の内積を求めなさい。}{dot_product_in_rectangle}
\QQRow{$\overrightarrow{\mathrm{AB}}\cdot\overrightarrow{\mathrm{AC}}$}{$36$}{$\overrightarrow{\mathrm{AD}}\cdot\overrightarrow{\mathrm{AC}}$}{$12$}
\QQRow{$\overrightarrow{\mathrm{BA}}\cdot\overrightarrow{\mathrm{AC}}$}{$-36$}{$\overrightarrow{\mathrm{AD}}\cdot\overrightarrow{\mathrm{CA}}$}{$-12$}

\TypeQuestionSingle{次の2つのベクトルの内積を求めなさい。}{component_dot_product}
\QQRow{$(2,3),\ (4,-1)$}{$5$}{$(-5,2),\ (1,6)$}{$7$}
\QQRow{$(3,-4),\ (-2,-7)$}{$22$}{$(\sqrt3,1),\ (\sqrt3,-1)$}{$2$}
\QQRow{$(0,8),\ (-3,5)$}{$40$}{$(a,b),\ (c,d)$}{$ac+bd$}

\TypeQuestionDouble{$\vec{a}=(2,-1)$、$\vec{b}=(-3,4)$ とする。次の値を求めなさい。}{component_dot_composite}
\QQRow{$(\vec{a}+\vec{b})\cdot\vec{a}$}{$-5$}{$|3\vec{a}-\vec{b}|^2$}{$130$}
\QQRow{$|\vec{a}+2\vec{b}|^2$}{$65$}{$(2\vec{a}-\vec{b})\cdot\vec{b}$}{$-45$}

\TypeQuestionDouble{$|\vec{a}|=2$、$|\vec{b}|=3$、$\vec{a}\cdot\vec{b}=1$ とする。次の内積を求めなさい。}{dot_product_expansion}
\QQRow{$\vec{a}\cdot(2\vec{a}-\vec{b})$}{$7$}{$\vec{b}\cdot(3\vec{a}+\vec{b})$}{$12$}
\QQRow{$(\vec{a}+\vec{b})\cdot(\vec{a}-\vec{b})$}{$-5$}{$(2\vec{a}-\vec{b})\cdot(\vec{a}+3\vec{b})$}{$-14$}
\QQRow{$|\vec{a}+\vec{b}|^2$}{$15$}{$|2\vec{a}-\vec{b}|^2$}{$21$}

\TypeQuestionSingle{次の条件から、$\vec{a}\cdot\vec{b}$ を求めなさい。}{recover_dot_product}
\QQRow{$|\vec{a}|=3,\ |\vec{b}|=4,\ |\vec{a}+\vec{b}|=5$}{$0$}{$|\vec{a}|=2,\ |\vec{b}|=5,\ |\vec{a}-\vec{b}|=\sqrt{33}$}{$-2$}
\QQRow{$|\vec{a}+\vec{b}|=6,\ |\vec{a}-\vec{b}|=4$}{$5$}{$|2\vec{a}+\vec{b}|^2-|2\vec{a}-\vec{b}|^2=24$}{$3$}

\TypeQuestionSingle{内積の性質を用いて、次の式を展開しなさい。}{dot_product_identity}
\QQRow{$|\vec{a}+\vec{b}|^2$}{$|\vec{a}|^2+2\vec{a}\cdot\vec{b}+|\vec{b}|^2$}{$|\vec{a}-\vec{b}|^2$}{$|\vec{a}|^2-2\vec{a}\cdot\vec{b}+|\vec{b}|^2$}
\QQRow{$(\vec{a}+2\vec{b})\cdot(\vec{a}-\vec{b})$}{$|\vec{a}|^2+\vec{a}\cdot\vec{b}-2|\vec{b}|^2$}{$|2\vec{a}-3\vec{b}|^2$}{$4|\vec{a}|^2-12\vec{a}\cdot\vec{b}+9|\vec{b}|^2$}

\TypeQuestionDouble{$|\vec{a}|=3$、$|\vec{b}|=4$、$\vec{a}\cdot\vec{b}=6$ とする。次のベクトルの大きさを求めなさい。}{magnitude_from_dot_product}
\QQRow{$|\vec{a}+\vec{b}|$}{$\sqrt{37}$}{$|\vec{a}-\vec{b}|$}{$\sqrt{13}$}
\QQRow{$|2\vec{a}+\vec{b}|$}{$2\sqrt{19}$}{$|2\vec{a}-3\vec{b}|$}{$6\sqrt3$}

\TypeQuestionDouble{$|\vec{a}+\vec{b}|$、$|\vec{a}-\vec{b}|$ が次のように与えられている。$|\vec{a}|^2+|\vec{b}|^2$ と $\vec{a}\cdot\vec{b}$ を求めなさい。}{sum_difference_norms}
\QQRow{$|\vec{a}+\vec{b}|=5,\quad |\vec{a}-\vec{b}|=3$}{$|\vec{a}|^2+|\vec{b}|^2=17,\ \vec{a}\cdot\vec{b}=4$}{$|\vec{a}+\vec{b}|=3\sqrt2,\quad |\vec{a}-\vec{b}|=2\sqrt3$}{$|\vec{a}|^2+|\vec{b}|^2=15,\ \vec{a}\cdot\vec{b}=\dfrac32$}
