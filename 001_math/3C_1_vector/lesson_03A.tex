\MajorQuestionLesson{ベクトルの成分と計算}

\TypeQuestion{次のベクトルの和・差・実数倍を成分で表しなさい。}{component_operations}
\QQRow{$(2,-3)+(-5,4)$}{}{$\left(\dfrac72,1\right)-\left(\dfrac12,-3\right)$}{}
\QQRow{$\dfrac32(4,-2)$}{}{$2(-4,3)-(1,5)$}{}
\QQRow{$\dfrac12(6,-8)+(-2,7)$}{}{$3(1,2)-2(-3,4)$}{}

\TypeQuestion{ベクトルの相等を用いて、実数 $x,y$ を求めなさい。}{component_equality}
\QQRow{$(x+1,2y)=(5,-6)$}{}{$(3x-2,y+4)=(7,1)$}{}
\QQRow{$\dfrac12(x,y)=(-4,5)$}{}{$(x-y,x+y)=(1,7)$}{}

\TypeQuestion{2点 $\mathrm{A},\mathrm{B}$ に対して、$\overrightarrow{\mathrm{AB}}$ の成分を求めなさい。}{points_to_components}
\QQRow{$\mathrm{A}(2,-1),\ \mathrm{B}(7,3)$}{}{$\mathrm{A}(-4,5),\ \mathrm{B}(1,-2)$}{}
\QQRow{$\mathrm{A}\left(\dfrac12,-1\right),\ \mathrm{B}\left(\dfrac72,2\right)$}{}{$\mathrm{A}(-1,-3),\ \mathrm{B}(4,2)$}{}
\QQRow{$\mathrm{A}\left(-2,\dfrac52\right),\ \mathrm{B}\left(1,-\dfrac32\right)$}{}{$\mathrm{A}(5,-2),\ \mathrm{B}(2,4)$}{}

\TypeQuestion{$\vec{a}=(2,-1)$、$\vec{b}=(-3,4)$ とする。次のベクトルの成分を求めなさい。}{component_linear_operations}
\QQRowThree{$\vec{a}+\vec{b}$}{}{$2\vec{a}-3\vec{b}$}{}{$-3\vec{a}+2\vec{b}$}{}
\QQRowThree{$\dfrac12\vec{a}+\vec{b}$}{}{$\dfrac32\vec{a}+\vec{b}$}{}{$-\vec{a}-2\vec{b}$}{}
