\MajorQuestionLesson{ベクトルの平行と分解}
\SetRowSpace{5mm}

\TypeQuestionDouble{$\vec{a}\ne\vec{0}$ とする。次のベクトルが $\vec{a}$ と同じ向きか、反対向きか、零ベクトルかを答えなさい。}{direction_by_scalar}
\QQRowThree{$3\vec{a}$}{}{$-2\vec{a}$}{}{$\dfrac15\vec{a}$}{}
\QQRowThree{$-\dfrac32\vec{a}$}{}{$0\vec{a}$}{}{$8\vec{a}$}{}

\TypeQuestionDouble{$\vec{a}\ne\vec{0}$ とし、$\vec{b}=k\vec{a}$ とする。次の条件から実数 $k$ を求めなさい。}{parallel_scalar}
\QQRow{$\vec{b}$ は $\vec{a}$ と同じ向きで、$|\vec{b}|=4|\vec{a}|$}{}{$\vec{b}$ は $\vec{a}$ と反対向きで、$|\vec{b}|=3|\vec{a}|$}{}
\QQRow{$\vec{a}$ は $\vec{b}$ と同じ向きで、$|\vec{a}|=2|\vec{b}|$}{}{$\vec{a}$ は $\vec{b}$ と反対向きで、$3|\vec{a}|=|\vec{b}|$}{}

\TypeQuestionDouble{$\vec{a},\vec{b}$ は平行でないベクトルとする。次のベクトルを $\vec{a},\vec{b}$ の係数を整理して表しなさい。}{collect_basis}
\QQRow{$2(\vec{a}-\vec{b})+3(\vec{a}+\vec{b})$}{}{$4(2\vec{a}+\vec{b})-(\vec{a}-3\vec{b})$}{}
\QQRow{$-3(\vec{a}+2\vec{b})+2(2\vec{a}-\vec{b})$}{}{$\dfrac12(6\vec{a}-4\vec{b})-2(\vec{a}+\vec{b})$}{}

\TypeQuestionDouble{$\vec{a},\vec{b}$ は平行でないベクトルで、$\vec{u}=2\vec{a}-\vec{b}$、$\vec{v}=-\vec{a}+3\vec{b}$ とする。次のベクトルを $\vec{a},\vec{b}$ を用いて表しなさい。}{change_of_generators}
\QQRow{$\vec{u}+\vec{v}$}{}{$2\vec{u}-\vec{v}$}{}
\QQRow{$3\vec{u}+2\vec{v}$}{}{$-\vec{u}+4\vec{v}$}{}
\QQRow{$\dfrac12\vec{u}-\vec{v}$}{}{$2(\vec{u}+\vec{v})-3\vec{u}$}{}

\TypeQuestionDouble{$\vec{a},\vec{b}$ は平行でないベクトルで、$\vec{x}=s\vec{a}+t\vec{b}$ とする。次の等式から実数 $s,t$ を求めなさい。}{coefficient_comparison}
\QQRow{$\vec{x}=3\vec{a}-2\vec{b}$}{}{$2\vec{x}=\vec{a}+6\vec{b}$}{}
\QQRow{$\vec{x}+\vec{a}=4\vec{a}-5\vec{b}$}{}{$3\vec{x}-2\vec{b}=6\vec{a}+\vec{b}$}{}

\TypeQuestionDouble{$\vec{a},\vec{b}$ は平行でないベクトルとする。次の2つのベクトルが平行になるように、実数 $k$ を求めなさい。}{parallel_condition_abstract}
\QQRow{$2\vec{a}-3\vec{b}$ と $4\vec{a}+k\vec{b}$}{}{$\vec{a}+2\vec{b}$ と $k\vec{a}-6\vec{b}$}{}
\QQRow{$3\vec{a}+k\vec{b}$ と $-2\vec{a}+4\vec{b}$}{}{$k\vec{a}-\vec{b}$ と $6\vec{a}+3\vec{b}$}{}

\TypeQuestionDouble{三角形 $\mathrm{ABC}$ で $\overrightarrow{\mathrm{AB}}=\vec{a}$、$\overrightarrow{\mathrm{AC}}=\vec{b}$ とする。条件を満たす点について、指定されたベクトルを $\vec{a},\vec{b}$ で表しなさい。}{parallel_geometry}
\QQRow{$\overrightarrow{\mathrm{AP}}=2\overrightarrow{\mathrm{AB}}$ のとき $\overrightarrow{\mathrm{CP}}$}{}{$\overrightarrow{\mathrm{AQ}}=-\overrightarrow{\mathrm{AC}}$ のとき $\overrightarrow{\mathrm{BQ}}$}{}
\QQRow{$\overrightarrow{\mathrm{AR}}=3\overrightarrow{\mathrm{BC}}$ のとき $\overrightarrow{\mathrm{AR}}$}{}{$\overrightarrow{\mathrm{BS}}=2\overrightarrow{\mathrm{CA}}$ のとき $\overrightarrow{\mathrm{AS}}$}{}
