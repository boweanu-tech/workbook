  \TextTall{四角形 ${\rm ABCD}$ は円に内接しており、$\Lseg{AB}=3$、$\Lseg{BC}=8$、$\Ang{ABC}=60^\circ$ である。対角線 $\Lseg{AC}$ を引き、$\Ang{ADC}$ の二等分線と $\Lseg{AC}$ との交点を ${\rm E}$ とする。また、$\Lseg{AE}:\Lseg{EC}=1:2$ である。図は必ずしも正確ではない。}

  \QQ{$\Lseg{AC}$ の長さを求めなさい。}{$7$}

  \QQ{四角形 ${\rm ABCD}$ の外接円の半径を求めなさい。}{$\dfrac{7\sqrt{3}}{3}$}
  \QQ{$\Ang{ADC}$ の大きさを求めなさい。}{$120^\circ$}
  \QQ{$\Lseg{AD}:\Lseg{DC}$ を最も簡単な整数の比で表しなさい。}{$1:2$}
  \QQ{$\Lseg{AD}$、$\Lseg{DC}$ の長さをそれぞれ求めなさい。}{$\Lseg{AD}=\sqrt{7},\quad \Lseg{DC}=2\sqrt{7}$}
  \QQ{$\cos\Ang{DAC}$ の値を求めなさい。}{$\dfrac{2\sqrt{7}}{7}$}
  \QQ{$\Lseg{DE}$ の長さを求めなさい。}{$\dfrac{2\sqrt{7}}{3}$}
  \QQ{四角形 ${\rm ABCD}$ の面積を求めなさい。}{$\dfrac{19\sqrt{3}}{2}$}
