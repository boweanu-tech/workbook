\MajorQuestion{円に内接する四角形の総合問題}
\SetRowSpace{4mm}

\FigProblemBlock{58mm}{fig/snipe_01_01}{
  \TextTall{四角形 ${\rm ABCD}$ は円に内接しており、$\Lseg{AB}=3$、$\Lseg{BC}=8$、$\Ang{ABC}=60^\circ$ である。対角線 $\Lseg{AC}$ を引き、$\Ang{ADC}$ の二等分線と $\Lseg{AC}$ との交点を ${\rm E}$ とする。また、$\Lseg{AE}:\Lseg{EC}=1:2$ である。図は必ずしも正確ではない。}
}{
  \QQ{$\cos\Ang{ABC}$ と $\sin\Ang{ABC}$ の値をそれぞれ求めなさい。}{$\cos\Ang{ABC}=\dfrac{1}{2},\quad \sin\Ang{ABC}=\dfrac{\sqrt{3}}{2}$}
  \QQ{$\Lseg{AC}$ の長さを求めなさい。}{$7$}
  \QQ{$\sin\Ang{ACB}$ の値を求めなさい。}{$\dfrac{3\sqrt{3}}{14}$}
  \QQ{$\cos\Ang{ACB}$ と $\tan\Ang{ACB}$ の値をそれぞれ求めなさい。}{$\cos\Ang{ACB}=\dfrac{13}{14},\quad \tan\Ang{ACB}=\dfrac{3\sqrt{3}}{13}$}
  \QQ{四角形 ${\rm ABCD}$ の外接円の半径を求めなさい。}{$\dfrac{7\sqrt{3}}{3}$}
  \QQ{$\sin\Ang{ADB}$ の値を求めなさい。}{$\dfrac{3\sqrt{3}}{14}$}
  \QQ{$\Ang{ADC}$ の大きさを求めなさい。}{$120^\circ$}
  \QQ{$\Lseg{AE}$、$\Lseg{EC}$ の長さをそれぞれ求めなさい。}{$\Lseg{AE}=\dfrac{7}{3},\quad \Lseg{EC}=\dfrac{14}{3}$}
  \QQ{$\Lseg{AD}:\Lseg{DC}$ を最も簡単な整数の比で表しなさい。}{$1:2$}
  \QQ{$\Lseg{AD}$、$\Lseg{DC}$ の長さをそれぞれ求めなさい。}{$\Lseg{AD}=\sqrt{7},\quad \Lseg{DC}=2\sqrt{7}$}
  \QQ{$\cos\Ang{DAC}$ の値を求めなさい。}{$\dfrac{2\sqrt{7}}{7}$}
  \QQ{$\Lseg{DE}$ の長さを求めなさい。}{$\dfrac{2\sqrt{7}}{3}$}
  \QQ{$\Tri{ABC}$ の面積を求めなさい。}{$6\sqrt{3}$}
  \QQ{$\Tri{ADC}$ の面積を求めなさい。}{$\dfrac{7\sqrt{3}}{2}$}
  \QQ{四角形 ${\rm ABCD}$ の面積を求めなさい。}{$\dfrac{19\sqrt{3}}{2}$}
  \QQ{四角形 ${\rm ABCD}$ の周の長さを求めなさい。}{$11+3\sqrt{7}$}
}
