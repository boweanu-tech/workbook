% 長方形と定数t
\TypeQuestion{}{figure_application_rectangle_square}
\FigProblemBlock{45mm}{fig/lesson_06_rectangle_symmetry}{\TextTall{図のように、放物線 $y=2x^2$ 上に、$x$ 座標が正である点 $\mathrm{P}$ がある。点 $\mathrm{P}$ の $x$ 座標を $t$ とし、点 $\mathrm{Q}$ は点 $\mathrm{P}$ と $y$ 軸について対称である。2点 $\mathrm{P},\mathrm{Q}$ から $x$ 軸へ引いた垂線と $x$ 軸との交点を、それぞれ $\mathrm{A},\mathrm{C}$ とする。また、点 $\mathrm{P}$ から $y$ 軸へ引いた垂線と $y$ 軸との交点を $\mathrm{B}$ とする。次の問いに答えなさい。}
\QQ{点 $\mathrm{P},\mathrm{Q}$ の座標を、それぞれ $t$ を用いて表しなさい。}{$\mathrm{P}(t,\ 2t^2),\ \mathrm{Q}(-t,\ 2t^2)$}
\QQ{線分 $\mathrm{OA},\mathrm{AP}$ の長さを、それぞれ $t$ を用いて表しなさい。}{$\mathrm{OA}=t,\ \mathrm{AP}=2t^2$}
}{
\QQ{$\mathrm{AP}=3\mathrm{OA}$ となるとき、$t$ の値を求めなさい。}{$t=\dfrac{3}{2}$}
\QQ{四角形 $\mathrm{OAPB}$ が正方形となるとき、$t$ の値を求めなさい。}{$t=\dfrac{1}{2}$}
\QQ{四角形 $\mathrm{CAPQ}$ が正方形となるとき、$t$ の値を求めなさい。}{$t=1$}
}

% 原点を頂点としない三角形と等積変形

\QQ{直線 $\mathrm{CE}$ の式を求めなさい。}{$y=-2x+\dfrac{3}{2}$}
\QQ{点 $\mathrm{E}$ の座標を求めなさい。}{$\left(\dfrac{3}{4},\ 0\right)$}
