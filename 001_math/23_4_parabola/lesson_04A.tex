\MajorQuestionLesson{放物線と直線}
\SetRowSpace{5mm}

\TypeQuestionSingle{次の放物線と直線の交点の座標を求めなさい。}{kouten}
\QQRow{$y=x^2$ と $y=2x+3$}{}{$y=x^2$ と $y=\dfrac{1}{2}x+\dfrac{3}{2}$}{}
\QQRow{$y=\dfrac{1}{2}x^2$ と $y=-x+4$}{}{$y=-\dfrac{1}{2}x^2$ と $y=\dfrac{1}{2}x-3$}{}
\QQRow{$y=3x^2$ と $y=\dfrac{3}{2}x+\dfrac{3}{2}$}{}{$y=-2x^2$ と $y=2x-4$}{}

\TypeQuestionSingle{放物線上にある2点A，Bの $x$ 座標が次のように与えられている。直線ABの式を求めなさい。}{chokusen}
\QQ{$y=x^2$ 上の点A，Bの $x$ 座標がそれぞれ $-2,\ 3$ である。}{}
\QQ{$y=\dfrac{1}{2}x^2$ 上の点A，Bの $x$ 座標がそれぞれ $-4,\ 2$ である。}{}
\QQ{$y=-2x^2$ 上の点A，Bの $x$ 座標がそれぞれ $-\dfrac{3}{2},\ \dfrac{1}{2}$ である。}{}
\QQ{$y=3x^2$ 上の点A，Bの $x$ 座標がそれぞれ $-\dfrac{1}{2},\ \dfrac{3}{2}$ である。}{}

\TypeQuestionSingle{次の問いに答えなさい。}{mojikettei}
\QQ{$y=ax^2$ と $y=3x+2$ が，$x=2$ の点で交わる。$a$ を求めなさい。}{}
\QQ{放物線 $y=x^2$ と，$y$ 切片が $3$ である直線が，$y$ 座標が $4$，$x$ 座標が正である点で交わる。この直線の式を求めなさい。}{}
\QQ{放物線 $y=-x^2$ と，$y$ 切片が $-3$ である直線が，$x=-2$ の点で交わる。この直線の式を求めなさい。}{}
\QQ{放物線 $y=\dfrac{1}{2}x^2$ と，傾きが $-2$ である直線が，$y$ 座標が $8$，$x$ 座標が正である点で交わる。この直線の式を求めなさい。}{}
