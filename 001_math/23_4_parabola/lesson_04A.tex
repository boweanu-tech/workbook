\MajorQuestionLesson{放物線と直線}
\SetRowSpace{5mm}

\TypeQuestionSingle{次の放物線と直線の交点の座標を求めなさい。}{kouten}
%直線の変化の割合や切片にも、分数が入る問題を混ぜて
\QQRow{$y=x^2$ と $y=2x+3$}{}{$y=x^2$ と $y=-2x+3$}{}
\QQRow{$y=\dfrac{1}{2}x^2$ と $y=-x+4$}{}{$y=-\dfrac{1}{2}x^2$ と $y=x-4$}{}
\QQRow{$y=3x^2$ と $y=6x+9$}{}{$y=-2x^2$ と $y=4x-6$}{}

\TypeQuestionSingle{放物線上にある2点A，Bの $x$ 座標が次のように与えられている。直線ABの式を求めなさい。}{chokusen}
%x座標にも、分数が入る問題を混ぜて
\QQ{$y=x^2$ 上の点A，Bの $x$ 座標がそれぞれ $-2,\ 3$ である。}{}
\QQ{$y=\dfrac{1}{2}x^2$ 上の点A，Bの $x$ 座標がそれぞれ $-4,\ 2$ である。}{}
\QQ{$y=-2x^2$ 上の点A，Bの $x$ 座標がそれぞれ $-2,\ 1$ である。}{}
\QQ{$y=3x^2$ 上の点A，Bの $x$ 座標がそれぞれ $-1,\ 2$ である。}{}

\TypeQuestionSingle{次の問いに答えなさい。}{mojikettei}
%指定されたy座標とか、「切片が3の直線と〜」とか表現の幅を広げて。
\QQ{$y=ax^2$ と $y=3x+2$ が，$x=2$ の点で交わる。$a$ を求めなさい。}{}
\QQ{$y=x^2$ と $y=mx+3$ が，$x=1$ の点で交わる。$m$ を求めなさい。}{}
\QQ{$y=-x^2$ と $y=3x+b$ が，$x=-2$ の点で交わる。$b$ を求めなさい。}{}
\QQ{$y=\dfrac{1}{2}x^2$ と $y=-2x+b$ が，$x=4$ の点で交わる。$b$ を求めなさい。}{}
