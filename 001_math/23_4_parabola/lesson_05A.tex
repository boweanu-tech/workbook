\MajorQuestionLesson{直線との利用問題}
\SetRowSpace{4mm}

\TypeQuestion{}{line_application_coefficients}
\FigProblemBlock{45mm}{fig/lesson_05_coefficients}{\Text{図のように、放物線 $y=ax^2$ と直線 $\ell:y=mx+12$ が2点 $\mathrm{A},\mathrm{B}$ で交わっている。点 $\mathrm{A}$ の座標は $(-2,4)$ であり、$a$は正の定数である。次の問いに答えなさい。}
\QQ{定数 $a$ の値を求めなさい。}{$a=1$}
\QQ{定数 $m$ の値を求めなさい。}{$m=4$}
}{
\QQ{点 $\mathrm{B}$ の座標を求めなさい。}{$(6,\ 36)$}
\QQ{$x$ の値が点 $\mathrm{A}$ の $x$ 座標から点 $\mathrm{B}$ の $x$ 座標まで増加するとき、放物線における変化の割合を求めなさい。}{$4$}
\QQ{$\Tri{OAB}$ の面積を求めなさい。}{$48$}
}

\TypeQuestion{}{line_application_axis_intercepts}
\FigProblemBlock{45mm}{fig/lesson_05_axis_intercepts}{\TextTall{図のように、放物線 $y=-\dfrac{1}{2}x^2$ と、2点 $\mathrm{P}(0,-4),\mathrm{Q}(4,0)$ を通る直線 $\ell$ が2点 $\mathrm{A},\mathrm{B}$ で交わっている。点 $\mathrm{A}$ の $x$ 座標は負、点 $\mathrm{B}$ の $x$ 座標は正である。次の問いに答えなさい。}
}{
\QQ{直線 $\ell$ の式を求めなさい。}{$y=x-4$}
\QQ{点 $\mathrm{A},\mathrm{B}$ の座標をそれぞれ求めなさい。}{$\mathrm{A}(-4,\ -8),\ \mathrm{B}(2,\ -2)$}
\QQ{$x$ の値が点 $\mathrm{A}$ の $x$ 座標から点 $\mathrm{B}$ の $x$ 座標まで増加するとき、放物線における変化の割合を求めなさい。}{$1$}
\QQ{$\Tri{OAB}$ の面積を求めなさい。}{$12$}
\QQ{$\Tri{OAP}$ と $\Tri{OPB}$ の面積比を、最も簡単な整数の比で求めなさい。}{$2:1$}
}

\TypeQuestion{}{line_application_two_lines}
\FigProblemBlock{45mm}{fig/lesson_05_two_lines}{\TextTall{図のように、放物線 $y=\dfrac{1}{2}x^2$ 上に点 $\mathrm{A}(-4,8)$ がある。また、同じ放物線上に2点 $\mathrm{B}(p,2),\mathrm{C}(q,8)$ があり、$p>0,\ q>0$ である。2点 $\mathrm{A},\mathrm{B}$ を通る直線を $\ell$、2点 $\mathrm{B},\mathrm{C}$ を通る直線を $m$ とする。また、直線 $\ell,m$ と $y$ 軸との交点をそれぞれ $\mathrm{D},\mathrm{E}$ とする。次の問いに答えなさい。}}{
\QQ{$p,q$ の値を求めなさい。}{$p=2,\ q=4$}
\QQ{直線 $\ell,m$ の式をそれぞれ求めなさい。}{$\ell:y=-x+4,\ m:y=3x-4$}
\QQ{$\Tri{BDE}$ の面積を求めなさい。}{$8$}
\QQ{$\Tri{ABC}$ の面積を求めなさい。}{$24$}
}

\TypeQuestion{}{line_application_chained_chords}
\FigProblemBlock{45mm}{fig/lesson_05_chained_chords}{\Text{図のように、放物線 $y=x^2$ 上に3点 $\mathrm{A},\mathrm{B},\mathrm{C}$ があり、それぞれの $x$ 座標は $-2,3,-3$ である。2点 $\mathrm{A},\mathrm{B}$ を通る直線 $\ell$ と $y$ 軸との交点を $\mathrm{P}$ とする。2点 $\mathrm{C},\mathrm{P}$ を通る直線を $m$ とし、直線 $m$ と放物線とのもう1つの交点を $\mathrm{D}$ とする。次の問いに答えなさい。}}{
\QQ{点 $\mathrm{A},\mathrm{B},\mathrm{C}$ の座標をそれぞれ求めなさい。}{$\mathrm{A}(-2,\ 4),\ \mathrm{B}(3,\ 9),\ \mathrm{C}(-3,\ 9)$}
\QQ{直線 $\ell$ の式と、点 $\mathrm{P}$ の座標を求めなさい。}{$\ell:y=x+6,\ \mathrm{P}(0,\ 6)$}
\QQ{直線 $m$ の式を求めなさい。}{$y=-x+6$}
\QQ{点 $\mathrm{D}$ の座標を求めなさい。}{$(2,\ 4)$}
}
