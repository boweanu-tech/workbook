\TypeQuestionSingle{次の関数で，指定された $x$ の値に対応する $y$ の値を順に答えなさい。}{atai}
\QQ{$y=2x^2$ で，$x=-3,\ -1,\ 0,\ 2,\ 4$}{$18,\ 2,\ 0,\ 8,\ 32$}
\QQ{$y=-x^2$ で，$x=-4,\ -2,\ 0,\ 1,\ 3$}{$-16,\ -4,\ 0,\ -1,\ -9$}
\QQ{$y=\dfrac{1}{2}x^2$ で，$x=-3,\ -1,\ 0,\ 2,\ 5$}{$\dfrac{9}{2},\ \dfrac{1}{2},\ 0,\ 2,\ \dfrac{25}{2}$}
\QQ{$y=-3x^2$ で，$x=-2,\ -1,\ \dfrac{1}{2},\ \dfrac{3}{2},\ 3$}{$-12,\ -3,\ -\dfrac{3}{4},\ -\dfrac{27}{4},\ -27$}

\LessonPageBreak

\TypeQuestionSingle{次の関数のグラフについて、各問いに答えなさい。}{gaikei}
\QQ{$y=x^2,\ y=3x^2,\ y=\dfrac{1}{2}x^2$ のうち，グラフの幅が最もせまいもの}{$y=3x^2$}
\QQ{$y=-2x^2,\ y=-\dfrac{1}{3}x^2,\ y=-5x^2$ のうち，グラフの幅が最も広いもの}{$y=-\dfrac{1}{3}x^2$}
\QQ{$y=2x^2,\ y=-2x^2,\ y=\dfrac{1}{2}x^2,\ y=-3x^2$ のうち，グラフの幅が同じもの}{$y=2x^2$ と $y=-2x^2$}
\QQ{$y=-x^2,\ y=2x^2,\ y=-\dfrac{1}{2}x^2,\ y=3x^2$ のうち，上に開くものすべて}{$y=2x^2,\ y=3x^2$}
\QQ{$y=2x^2,\ y=-3x^2,\ y=\dfrac{1}{4}x^2$ のうち，$x<0$ で $y$ が常に減少するものすべて}{$y=2x^2,\ y=\dfrac{1}{4}x^2$}
\QQ{$y=x^2,\ y=-2x^2,\ y=-\dfrac{1}{3}x^2$ のうち，$x>0$ で $y$ が常に減少するものすべて}{$y=-2x^2,\ y=-\dfrac{1}{3}x^2$}
\QQ{$y=-\dfrac{1}{2}x^2,\ y=-3x^2,\ y=2x^2$ のうち，下に開き，$y=-x^2$ より幅がせまいもの}{$y=-3x^2$}
\QQ{$y=\dfrac{1}{3}x^2,\ y=-2x^2,\ y=-\dfrac{1}{2}x^2$ のうち，$y=x^2$ より幅が広いものすべて}{$y=\dfrac{1}{3}x^2,\ y=-\dfrac{1}{2}x^2$}
