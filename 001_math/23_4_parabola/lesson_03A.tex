\MajorQuestionLesson{変域と変化の割合}
\SetRowSpace{6mm}

\TypeQuestionSingle{次の関数で，$x$ の変域が指定されているとき，$y$ の変域を求めなさい。}{domain_basic}
\QQRow{$y=x^2$，$-2\leqq x\leqq 3$}{}{$y=2x^2$，$-1\leqq x\leqq 4$}{}
\QQRow{$y=-x^2$，$-3\leqq x\leqq 2$}{}{$y=-2x^2$，$-4\leqq x\leqq 1$}{}
\QQRow{$y=\dfrac{1}{2}x^2$，$-6\leqq x\leqq -2$}{}{$y=-\dfrac{1}{3}x^2$，$2\leqq x\leqq 6$}{}
\QQRow{$y=3x^2$，$-4\leqq x\leqq -1$}{}{$y=-\dfrac{1}{2}x^2$，$-5\leqq x\leqq -1$}{}

\TypeQuestionSingle{次の関数で，$x$ の変域が指定されているとき，$y$ の最大値と最小値を求めなさい。}{maxmin}
\QQ{$y=x^2$，$-3\leqq x\leqq 1$}{}
\QQ{$y=2x^2$，$-2\leqq x\leqq 5$}{}
\QQ{$y=-x^2$，$-1\leqq x\leqq 4$}{}
\QQ{$y=-3x^2$，$-5\leqq x\leqq 2$}{}
\QQ{$y=\dfrac{1}{4}x^2$，$2\leqq x\leqq 8$}{}
\QQ{$y=-\dfrac{1}{2}x^2$，$-6\leqq x\leqq -2$}{}

\TypeQuestionSingle{次の関数で，$y$ の変域が指定された範囲になるように，$x$ の変域を求めなさい。ただし，$x$ の変域は $0\leqq x$ の範囲で考えるものとする。}{domain_inverse_positive}
\QQRow{$y=x^2$，$0\leqq y\leqq 25$}{}{$y=2x^2$，$0\leqq y\leqq 18$}{}
\QQRow{$y=\dfrac{1}{3}x^2$，$0\leqq y\leqq 12$}{}{$y=-x^2$，$-36\leqq y\leqq 0$}{}
\QQRow{$y=-2x^2$，$-50\leqq y\leqq 0$}{}{$y=-\dfrac{1}{4}x^2$，$-16\leqq y\leqq 0$}{}

\TypeQuestionSingle{次の関数で，$x$ の値が指定された範囲で変化するとき，変化の割合を求めなさい。}{rate_basic}
\QQRow{$y=x^2$，$x$ が $1$ から $4$ まで}{}{$y=2x^2$，$x$ が $-1$ から $3$ まで}{}
\QQRow{$y=-x^2$，$x$ が $2$ から $5$ まで}{}{$y=-3x^2$，$x$ が $-4$ から $1$ まで}{}
\QQRow{$y=\dfrac{1}{2}x^2$，$x$ が $-2$ から $6$ まで}{}{$y=-\dfrac{1}{3}x^2$，$x$ が $-6$ から $3$ まで}{}
\QQRow{$y=4x^2$，$x$ が $-3$ から $-1$ まで}{}{$y=-\dfrac{1}{2}x^2$，$x$ が $2$ から $8$ まで}{}

\TypeQuestionSingle{次の関数で，$x$ の値が指定された範囲で変化するとき，$x$ の増加量，$y$ の増加量，変化の割合を求めなさい。}{rate_detail}
\QQ{$y=x^2$，$x$ が $2$ から $5$ まで}{}
\QQ{$y=2x^2$，$x$ が $-3$ から $1$ まで}{}
\QQ{$y=-x^2$，$x$ が $-2$ から $4$ まで}{}
\QQ{$y=-\dfrac{1}{2}x^2$，$x$ が $-6$ から $2$ まで}{}

\TypeQuestionDouble{次の関数で，指定された $x$ の変域における $y$ の変域と変化の割合をそれぞれ求めなさい。}{domain_and_rate}
