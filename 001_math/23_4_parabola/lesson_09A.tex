\MajorQuestionLesson{点や図形の移動}
\SetRowSpace{3mm}

% 長方形上を動く2点
\TypeQuestion{}{moving_points_rectangle}
\FigProblemBlock{50mm}{fig/lesson_09_moving_rectangle}{\TextTall{図のように、$\Lseg{AB}=10\mathrm{cm}$、$\Lseg{AD}=12\mathrm{cm}$ の長方形 $\mathrm{ABCD}$ がある。2点 $\mathrm{P},\mathrm{Q}$ は同時に点 $\mathrm{A}$ を出発し、点 $\mathrm{P}$ は辺 $\mathrm{AB}$ 上を毎秒 $2\mathrm{cm}$ で点 $\mathrm{B}$ まで、点 $\mathrm{Q}$ は辺 $\mathrm{AD}$ 上を毎秒 $1\mathrm{cm}$ で点 $\mathrm{D}$ まで動き、それぞれ到着した点で止まる。出発してから $x$ 秒後の $\Tri{APQ}$ の面積を $y\mathrm{cm}^2$ とするとき、次の問いに答えなさい。ただし、$0\leqq x\leqq 12$ とする。}}{
\QQ{点 $\mathrm{P}$ が点 $\mathrm{B}$ に到着するときの $x$ と $y$ の値を求めなさい。}{$x=5,\ y=25$}
\QQ{$0\leqq x\leqq 5$ のとき、$y$ を $x$ の式で表しなさい。}{$y=x^2$}
\QQ{$5\leqq x\leqq 12$ のとき、$y$ を $x$ の式で表しなさい。}{$y=5x$}
\QQ{$x$ と $y$ の関係をグラフに表しなさい。}{$y=\begin{cases}x^2&(0\leqq x\leqq 5)\\5x&(5\leqq x\leqq 12)\end{cases}$ のグラフ}
\QQ{$y=40$ となるときの $x$ の値を求めなさい。}{$x=8$}
}

% 直角三角形の2辺上を動く2点
\TypeQuestion{}{moving_points_right_triangle}
\FigProblemBlock{50mm}{fig/lesson_09_moving_right_triangle}{\TextTall{図のように、$\Ang{BAC}=90^\circ$、$\Lseg{AB}=12\mathrm{cm}$、$\Lseg{AC}=8\mathrm{cm}$ の直角三角形 $\mathrm{ABC}$ がある。2点 $\mathrm{P},\mathrm{Q}$ は同時に点 $\mathrm{A}$ を出発し、点 $\mathrm{P}$ は辺 $\mathrm{AB}$ 上を毎秒 $3\mathrm{cm}$ で点 $\mathrm{B}$ まで、点 $\mathrm{Q}$ は辺 $\mathrm{AC}$ 上を毎秒 $1\mathrm{cm}$ で点 $\mathrm{C}$ まで動き、それぞれ到着した点で止まる。出発してから $x$ 秒後の $\Tri{APQ}$ の面積を $y\mathrm{cm}^2$ とするとき、次の問いに答えなさい。ただし、$0\leqq x\leqq 8$ とする。}}{
\QQ{$0\leqq x\leqq 4$ のとき、$y$ を $x$ の式で表しなさい。}{$y=\dfrac{3}{2}x^2$}
\QQ{$4\leqq x\leqq 8$ のとき、$y$ を $x$ の式で表しなさい。}{$y=6x$}
\QQ{$x$ と $y$ の関係をグラフに表しなさい。}{$y=\begin{cases}\dfrac{3}{2}x^2&(0\leqq x\leqq 4)\\6x&(4\leqq x\leqq 8)\end{cases}$ のグラフ}
\QQ{$y=18$ となるときと $y=42$ となるときの $x$ の値を、それぞれ求めなさい。}{$x=2\sqrt{3},\ 7$}
}

% 長方形と台形の重なり
\TypeQuestion{}{moving_overlap_trapezoid}
\FigProblemBlock{58mm}{fig/lesson_09_overlap_trapezoid}{\TextTall{図のように、横 $10\mathrm{cm}$、縦 $4\mathrm{cm}$ の長方形 $\mathrm{ABCD}$ と、その左側に台形 $\mathrm{EFGH}$ がある。$\mathrm{GH}\perp\mathrm{EH}$、$\Lseg{EH}\Parallel\Lseg{FG}$、$\Lseg{EH}=8\mathrm{cm}$、$\Lseg{FG}=4\mathrm{cm}$、$\Lseg{GH}=4\mathrm{cm}$ である。図は $0$ 秒後の状態を表している。台形は形と向きを変えずに毎秒 $1\mathrm{cm}$ で右に動く。$x$ 秒後に2つの図形が重なる部分の面積を $y\mathrm{cm}^2$ とするとき、次の問いに答えなさい。}}{
\QQ{$x=4$ のとき、$y$ の値を求めなさい。}{$y=8$}
\QQ{$0\leqq x\leqq 4$ のとき、$y$ を $x$ の式で表しなさい。}{$y=\dfrac{1}{2}x^2$}
\QQ{$4\leqq x\leqq 8$ のとき、$y$ を $x$ の式で表しなさい。}{$y=4x-8$}
\QQ{$0\leqq x\leqq 8$ の範囲で、$x$ と $y$ の関係をグラフに表しなさい。}{$y=\begin{cases}\dfrac{1}{2}x^2&(0\leqq x\leqq 4)\\4x-8&(4\leqq x\leqq 8)\end{cases}$ のグラフ}
\QQ{$0\leqq x\leqq 8$ の範囲で、$y=16$ となるときの $x$ の値を求めなさい。}{$x=6$}
}

% 長方形と直角二等辺三角形の重なり
\TypeQuestion{}{moving_overlap_right_isosceles_triangle}
\FigProblemBlock{58mm}{fig/lesson_09_overlap_triangle}{\TextTall{図のように、横 $8\mathrm{cm}$、縦 $6\mathrm{cm}$ の長方形 $\mathrm{ABCD}$ と、その左側に $\Lseg{QR}=8\mathrm{cm}$ の直角二等辺三角形 $\mathrm{PQR}$ がある。図は $0$ 秒後の状態を表している。三角形は形と向きを変えずに毎秒 $1\mathrm{cm}$ で右に動く。$x$ 秒後に2つの図形が重なる部分の面積を $y\mathrm{cm}^2$ とするとき、次の問いに答えなさい。}}{
\QQ{$0\leqq x\leqq 3$ のとき、$y$ を $x$ の式で表しなさい。}{$y=x^2$}
\QQ{$x=3$ のとき、$y$ の値を求めなさい。}{$y=9$}
\QQ{$3\leqq x\leqq 4$ のとき、$y$ を $x$ の式で表しなさい。}{$y=6x-9$}
\QQ{$4\leqq x\leqq 8$ のとき、$y$ を $x$ の式で表しなさい。}{$y=15$}
\QQ{$0\leqq x\leqq 8$ の範囲で、$x$ と $y$ の関係をグラフに表しなさい。}{$y=\begin{cases}x^2&(0\leqq x\leqq 3)\\6x-9&(3\leqq x\leqq 4)\\15&(4\leqq x\leqq 8)\end{cases}$ のグラフ}
\QQ{$0\leqq x\leqq 8$ の範囲で、$y=12$ となるときの $x$ の値を求めなさい。}{$x=\dfrac{7}{2}$}
}
