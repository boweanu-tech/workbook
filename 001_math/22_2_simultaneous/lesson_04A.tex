\TypeQuestionSingle{次の連立方程式を解きなさい。}{common_factor}
%50x+200y=80　のような、ゼロの数が違う数もまじるもの、
%10と同時に5や3なども一緒に割ったほうが早いもの(20x-80y=-440のような)なども混ぜて。
%あと2題追加
%答えはすべて整数に。
\QQRow{$\begin{cases}20x+30y=130\\50x-20y=40\end{cases}$}{}{$\begin{cases}30x+20y=-20\\40x-60y=-460\end{cases}$}{}
\QQRow{$\begin{cases}40x-70y=310\\30x+50y=130\end{cases}$}{}{$\begin{cases}500x+200y=-1600\\300x-400y=600\end{cases}$}{}
\QQRow{$\begin{cases}600x+300y=-900\\200x-500y=2700\end{cases}$}{}{$\begin{cases}70x+50y=-220\\20x-80y=-440\end{cases}$}{}

\TypeQuestionSingle{次の連立方程式を解きなさい。}{fraction}
%あと2題追加
%答えに分数がまじるものも1/3程度入れてください。
%複雑になりすぎないように。
\QQRow{$\begin{cases}\dfrac{x}{2}+\dfrac{y}{3}=1\\[1mm]\dfrac{x}{4}-\dfrac{y}{2}=\dfrac52\end{cases}$}{}{$\begin{cases}\dfrac{x}{3}+\dfrac{y}{5}=\dfrac13\\[1mm]\dfrac{x}{2}-\dfrac{y}{10}=-\dfrac32\end{cases}$}{}
\QQRow{$\begin{cases}\dfrac{x}{3}-\dfrac{y}{2}=3\\[1mm]\dfrac{x}{2}+\dfrac{y}{4}=\dfrac12\end{cases}$}{}{$\begin{cases}\dfrac{x}{5}+\dfrac{y}{2}=-2\\[1mm]\dfrac{x}{2}-\dfrac{y}{4}=-2\end{cases}$}{}
\QQRow{$\begin{cases}\dfrac{x}{3}+\dfrac{y}{2}=\dfrac52\\[1mm]\dfrac{x}{2}-\dfrac{y}{4}=\dfrac{11}{4}\end{cases}$}{}{$\begin{cases}\dfrac{x}{4}-\dfrac{y}{3}=\dfrac94\\[1mm]\dfrac{x}{2}+\dfrac{y}{6}=-\dfrac12\end{cases}$}{}
