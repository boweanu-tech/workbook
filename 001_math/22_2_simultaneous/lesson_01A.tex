\MajorQuestionLesson{連立方程式の意味と解}
\SetRowSpace{5mm}

\TypeQuestionDouble{次の数の組を二元一次方程式に代入したとき、等式が成り立つ場合は○、成り立たない場合は×を答えなさい。}{nigenkai}
\QQRow{$x=2,\ y=3$ は $x+y=5$ の解である}{}{$x=-1,\ y=4$ は $2x+y=1$ の解である}{}
\QQRow{$x=3,\ y=-2$ は $x-2y=6$ の解である}{}{$x=0,\ y=-5$ は $3x-y=5$ の解である}{}
\QQRow{$x=-2,\ y=-1$ は $2x+3y=-7$ の解である}{}{$x=4,\ y=1$ は $x+4y=9$ の解である}{}
\QQRow{$x=\dfrac{1}{2},\ y=2$ は $4x-y=0$ の解である}{}{$x=-3,\ y=\dfrac{1}{2}$ は $x+2y=-1$ の解である}{}

\TypeQuestionDouble{次の数の組が、与えられた連立方程式の解である場合は○、そうでない場合は×を答えなさい。}{kaihan}
\QQRow{$(x,y)=(2,1),\quad \begin{cases}x+y=3\\2x-y=3\end{cases}$}{}{$(x,y)=(1,2),\quad \begin{cases}x+y=3\\2x-y=3\end{cases}$}{}
\QQRow{$(x,y)=(-1,3),\quad \begin{cases}2x+y=1\\x-y=-3\end{cases}$}{}{$(x,y)=(3,-1),\quad \begin{cases}x-2y=5\\3x+y=8\end{cases}$}{}
\QQRow{$(x,y)=(0,-2),\quad \begin{cases}3x+y=-2\\x-2y=5\end{cases}$}{}{$(x,y)=(-2,-3),\quad \begin{cases}x+y=-5\\2x-y=-1\end{cases}$}{}
\QQRow{$(x,y)=\left(\dfrac{1}{2},1\right),\quad \begin{cases}2x+y=2\\4x-y=2\end{cases}$}{}{$(x,y)=\left(-1,\dfrac{1}{2}\right),\quad \begin{cases}x+2y=0\\3x-2y=-4\end{cases}$}{}

\TypeQuestionDouble{候補のうち、与えられた連立方程式の解である数の組を答えなさい。}{kouho}
\QQ{$(x,y)=(1,2),\ (2,1),\ (3,0)$ のうち、$\begin{cases}x+y=3\\2x+y=5\end{cases}$ の解であるもの}{}
\QQ{$(x,y)=(-1,2),\ (-2,1),\ (1,-2)$ のうち、$\begin{cases}x-y=-3\\3x+y=-1\end{cases}$ の解であるもの}{}
\QQ{$(x,y)=(0,-3),\ (1,-2),\ (2,-1)$ のうち、$\begin{cases}x-y=3\\2x+y=3\end{cases}$ の解であるもの}{}
\QQ{$(x,y)=(-2,-2),\ (-3,-1),\ (-1,-3)$ のうち、$\begin{cases}x+y=-4\\x-2y=2\end{cases}$ の解であるもの}{}

\LessonPageBreak

\TypeQuestionDouble{示された $x$ または $y$ の値と組み合わせたとき、連立方程式の解となるもう一方の値を求めなさい。}{katakai}
\QQRow{$\begin{cases}x+y=7\\2x-y=2\end{cases}$ で、$x=3$}{}{$\begin{cases}2x+y=-1\\x-y=-5\end{cases}$ で、$y=3$}{}
\QQRow{$\begin{cases}3x-y=11\\x+2y=6\end{cases}$ で、$x=4$}{}{$\begin{cases}x-3y=-8\\2x+y=-2\end{cases}$ で、$y=2$}{}
\QQRow{$\begin{cases}4x+y=3\\2x-3y=5\end{cases}$ で、$x=1$}{}{$\begin{cases}3x+2y=-4\\x-y=-3\end{cases}$ で、$y=1$}{}
\QQRow{$\begin{cases}2x+y=4\\3x-2y=-1\end{cases}$ で、$x=1$}{}{$\begin{cases}x+2y=0\\2x-y=-5\end{cases}$ で、$y=1$}{}

\TypeQuestionDouble{指定された数の組が連立方程式の解になるように、$a$ の値を求めなさい。}{ateisu}
\QQ{$x=2,\ y=1$ が $\begin{cases}x+y=3\\ax-y=5\end{cases}$ の解である}{}
\QQ{$x=-1,\ y=3$ が $\begin{cases}2x+y=1\\x+ay=5\end{cases}$ の解である}{}
\QQ{$x=3,\ y=-2$ が $\begin{cases}ax+y=10\\x-2y=7\end{cases}$ の解である}{}
\QQ{$x=-2,\ y=-1$ が $\begin{cases}3x-y=-5\\ax+2y=4\end{cases}$ の解である}{}
\QQ{$x=\dfrac{1}{2},\ y=2$ が $\begin{cases}4x-y=0\\2ax+y=5\end{cases}$ の解である}{}

\TypeQuestionDouble{指定された数の組が連立方程式の解になるように、$a,\ b$ の値を求めなさい。}{niteisu}
\QQ{$x=1,\ y=2$ が $\begin{cases}ax+y=5\\2x+by=8\end{cases}$ の解である}{}
\QQ{$x=-2,\ y=3$ が $\begin{cases}ax-y=5\\x+by=10\end{cases}$ の解である}{}
\QQ{$x=3,\ y=-1$ が $\begin{cases}2x+ay=4\\bx-y=10\end{cases}$ の解である}{}
\QQ{$x=-1,\ y=-2$ が $\begin{cases}ax+3y=-4\\2x+by=6\end{cases}$ の解である}{}
