\TypeQuestionDouble{次の数の組を二元一次方程式に代入したとき、等式が成り立つ場合は○、成り立たない場合は×を答えなさい。}{nigenkai}
\QQRow{$x=2,\ y=3$ は $x+y=5$ の解である}{}{$x=-1,\ y=4$ は $2x+y=1$ の解である}{}
\QQRow{$x=3,\ y=-2$ は $x-2y=6$ の解である}{}{$x=0,\ y=-5$ は $3x-y=5$ の解である}{}
\QQRow{$x=\dfrac{1}{2},\ y=2$ は $4x-y=0$ の解である}{}{$x=-3,\ y=\dfrac{1}{2}$ は $x+2y=-1$ の解である}{}

\TypeQuestionDouble{次の数の組が、与えられた連立方程式の解である場合は○、そうでない場合は×を答えなさい。}{kaihan}
\QQRow{$(x,y)=(-1,3),\quad \begin{cases}2x+y=1\\x-y=-3\end{cases}$}{}{$(x,y)=(3,-1),\quad \begin{cases}x-2y=5\\3x+y=8\end{cases}$}{}
\QQRow{$(x,y)=(0,-2),\quad \begin{cases}3x+y=-2\\x-2y=5\end{cases}$}{}{$(x,y)=(-2,-3),\quad \begin{cases}x+y=-5\\2x-y=-1\end{cases}$}{}
\QQRow{$(x,y)=\left(\dfrac{1}{2},1\right),\quad \begin{cases}2x+y=2\\4x-y=2\end{cases}$}{}{$(x,y)=(1,2),\quad \begin{cases}x+y=3\\2x-y=3\end{cases}$}{}

\TypeQuestionDouble{候補のうち、与えられた連立方程式の解である数の組を答えなさい。}{kouho}

\QQ{$(x,y)=(-1,2),\ (-2,1),\ (1,-2)$ のうち、$\begin{cases}x-y=-3\\3x+y=-1\end{cases}$ の解であるもの}{}

\TypeQuestionDouble{示された $x$ または $y$ の値と組み合わせたとき、連立方程式の解となるもう一方の値を求めなさい。}{katakai}
\QQRow{$\begin{cases}x+y=7\\2x-y=2\end{cases}$ で、$x=3$}{}{$\begin{cases}2x+y=-1\\x-y=-5\end{cases}$ で、$y=3$}{}
\QQRow{$\begin{cases}3x-y=11\\x+2y=6\end{cases}$ で、$x=4$}{}{$\begin{cases}x-3y=-8\\2x+y=-2\end{cases}$ で、$y=2$}{}
\QQRow{$\begin{cases}4x+y=3\\2x-3y=5\end{cases}$ で、$x=1$}{}{$\begin{cases}3x+2y=-4\\x-y=-3\end{cases}$ で、$y=1$}{}
\QQRow{$\begin{cases}2x+y=4\\3x-2y=-1\end{cases}$ で、$x=1$}{}{$\begin{cases}x+2y=0\\2x-y=-5\end{cases}$ で、$y=1$}{}
