\TypeQuestionDouble{次の数の組を二元一次方程式に代入したとき、等式が成り立つ場合は○、成り立たない場合は×を答えなさい。}{nigenkai}
\QQRow{$x=2,\ y=-2$ は $3x+2y=5$ の解である}{×}{$x=-2,\ y=3$ は $x-3y=-11$ の解である}{○}
\QQRow{$x=1,\ y=4$ は $2x-y=-2$ の解である}{○}{$x=0,\ y=-3$ は $4x-y=3$ の解である}{○}
\QQRow{$x=\dfrac{1}{2},\ y=3$ は $2x+3y=8$ の解である}{×}{$x=-4,\ y=\dfrac{1}{2}$ は $x+2y=0$ の解である}{×}

\TypeQuestionDouble{次の数の組が、与えられた連立方程式の解である場合は○、そうでない場合は×を答えなさい。}{kaihan}
\QQRow{$(x,y)=(1,-2),\quad \begin{cases}2x-y=4\\x+3y=-5\end{cases}$}{○}{$(x,y)=(-3,1),\quad \begin{cases}x-y=-7\\2x+y=-5\end{cases}$}{×}

\TypeQuestionSingle{候補のうち、与えられた連立方程式の解である数の組を答えなさい。}{kouho}
\QQ{$(x,y)=(2,1),\ (4,-1),\ (-2,3)$ のうち、$\begin{cases}x+y=3\\2x-y=9\end{cases}$ の解であるもの}{$(x,y)=(4,-1)$}
\QQ{$(x,y)=(1,-3),\ (-2,0),\ (-1,-3)$ のうち、$\begin{cases}x-2y=5\\x+4y=-13\end{cases}$ の解であるもの}{$(x,y)=(-1,-3)$}
\QQ{$(x,y)=(0,5),\ (2,0),\ (-1,4)$ のうち、$\begin{cases}2x+y=5\\3x-y=-5\end{cases}$ の解であるもの}{$(x,y)=(0,5)$}
\QQ{$(x,y)=(1,3),\ (3,2),\ (4,0)$ のうち、$\begin{cases}x+2y=7\\2x-3y=0\end{cases}$ の解であるもの}{$(x,y)=(3,2)$}

\TypeQuestionDouble{示された $x$ または $y$ の値と組み合わせたとき、連立方程式の解となるもう一方の値を求めなさい。}{katakai}
\QQRow{$\begin{cases}x+2y=-4\\3x-y=9\end{cases}$ で、$x=2$}{$y=-3$}{$\begin{cases}2x+y=2\\x-3y=-13\end{cases}$ で、$x=-1$}{$y=4$}
\QQRow{$\begin{cases}x+y=6\\2x-3y=7\end{cases}$ で、$y=1$}{$x=5$}{$\begin{cases}3x+2y=5\\x-4y=11\end{cases}$ で、$x=3$}{$y=-2$}
\QQRow{$\begin{cases}3x+y=-6\\2x-y=-14\end{cases}$ で、$y=6$}{$x=-4$}{$\begin{cases}x-2y=17\\4x+y=23\end{cases}$ で、$y=-5$}{$x=7$}
