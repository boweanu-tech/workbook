\TypeQuestion{次の連立方程式を代入法で解きなさい。}{substitute_y}
\QQRow{$\begin{cases}y=2x-5\\3x+y=10\end{cases}$}{}{$\begin{cases}y=-3x+4\\2x-y=-9\end{cases}$}{}
\QQRow{$\begin{cases}y=x+6\\4x-3y=-20\end{cases}$}{}{$\begin{cases}y=5-2x\\3x+4y=5\end{cases}$}{}
\QQRow{$\begin{cases}y=-x-4\\5x-2y=8\end{cases}$}{}{$\begin{cases}y=4x+1\\7x-2y=-4\end{cases}$}{}
\QQRow{$\begin{cases}y=3x-8\\2x+5y=28\end{cases}$}{}{$\begin{cases}y=6-x\\3x-4y=11\end{cases}$}{}

\TypeQuestion{次の連立方程式を代入法で解きなさい。}{substitute_x}

\QQRow{$\begin{cases}x=-y-4\\3x-2y=-17\end{cases}$}{}{$\begin{cases}x=4y-3\\2x+5y=-19\end{cases}$}{}
\QQRow{$\begin{cases}x=2-y\\5x+3y=14\end{cases}$}{}{$\begin{cases}x=3-5y\\2x-y=17\end{cases}$}{}
\QQRow{$\begin{cases}x=2y+6\\3x+4y=-12\end{cases}$}{}{$\begin{cases}x=y-7\\2x-3y=-18\end{cases}$}{}

\TypeQuestion{次の連立方程式を代入法で解きなさい。係数が $1$ または $-1$ である文字に着目しなさい。}{isolate_then_substitute}
\QQRow{$\begin{cases}x+y=7\\3x-2y=1\end{cases}$}{}{$\begin{cases}x-2y=-8\\4x+y=-5\end{cases}$}{}
\QQRow{$\begin{cases}3x+y=-11\\2x-5y=21\end{cases}$}{}{$\begin{cases}2x-y=9\\5x+3y=6\end{cases}$}{}
\QQRow{$\begin{cases}x+4y=-7\\3x-y=18\end{cases}$}{}{$\begin{cases}5x-y=18\\2x+3y=14\end{cases}$}{}
\QQRow{$\begin{cases}x-3y=13\\4x+y=0\end{cases}$}{}{$\begin{cases}2x+y=-1\\3x-4y=-29\end{cases}$}{}
\QQRow{$\begin{cases}x+y=-6\\5x-2y=-16\end{cases}$}{}{$\begin{cases}-x+2y=9\\5x+y=-1\end{cases}$}{}
