\MajorQuestionLesson{代入法}

\TypeQuestion{次の連立方程式を代入法で解きなさい。}{direct_substitution}
\QQRow{$\begin{cases}y=2x-5\\3x+y=10\end{cases}$}{$x=3,\ y=1$}{$\begin{cases}y=-3x+4\\2x-y=-9\end{cases}$}{$x=-1,\ y=7$}
\QQRow{$\begin{cases}3x-2y=-17\\x=-y-4\end{cases}$}{$x=-5,\ y=1$}{$\begin{cases}y=5-2x\\3x+4y=5\end{cases}$}{$x=3,\ y=-1$}
\QQRow{$\begin{cases}2x-y=12\\x=3y+1\end{cases}$}{$x=7,\ y=2$}{$\begin{cases}7x-2y=-4\\y=4x+1\end{cases}$}{$x=2,\ y=9$}
\QQRow{$\begin{cases}x=4y-3\\2x+5y=-19\end{cases}$}{$x=-7,\ y=-1$}{$\begin{cases}4x-3y=-20\\y=x+6\end{cases}$}{$x=-2,\ y=4$}
\QQRow{$\begin{cases}x=2-y\\5x+3y=14\end{cases}$}{$x=4,\ y=-2$}{$\begin{cases}3x-4y=11\\y=6-x\end{cases}$}{$x=5,\ y=1$}
\QQRow{$\begin{cases}x=3-5y\\2x-y=17\end{cases}$}{$x=8,\ y=-1$}{$\begin{cases}3x+4y=-12\\x=2y+6\end{cases}$}{$x=0,\ y=-3$}

\TypeQuestion{次の連立方程式を代入法で解きなさい。係数が $1$ または $-1$ である文字に着目しなさい。}{isolate_then_substitute}
\QQRow{$\begin{cases}x+y=7\\3x-2y=1\end{cases}$}{$x=3,\ y=4$}{$\begin{cases}3x+y=-11\\2x-5y=21\end{cases}$}{$x=-2,\ y=-5$}
\QQRow{$\begin{cases}2x-y=9\\5x+3y=6\end{cases}$}{$x=3,\ y=-3$}{$\begin{cases}5x-y=18\\2x+3y=14\end{cases}$}{$x=4,\ y=2$}
\QQRow{$\begin{cases}x-3y=13\\4x+y=0\end{cases}$}{$x=1,\ y=-4$}{$\begin{cases}-x+2y=9\\5x+y=-1\end{cases}$}{$x=-1,\ y=4$}

\TypeQuestion{次の連立方程式を代入法で解きなさい。}{substitute_multiples}
\QQRow{$\begin{cases}2x=y+7\\2x+3y=11\end{cases}$}{$x=4,\ y=1$}{$\begin{cases}4x-3y=-13\\3y=x+10\end{cases}$}{$x=-1,\ y=3$}
\QQRow{$\begin{cases}3x-4y=-17\\3x=y-11\end{cases}$}{$x=-3,\ y=2$}{$\begin{cases}4y=2x+8\\3x-4y=-4\end{cases}$}{$x=4,\ y=4$}
\QQRow{$\begin{cases}5x=4y+3\\2y+5x=21\end{cases}$}{$x=3,\ y=3$}{$\begin{cases}6x-2y=26\\2y=9-x\end{cases}$}{$x=5,\ y=2$}
\QQRow{$\begin{cases}3x=2y-8\\5y-3x=11\end{cases}$}{$x=-2,\ y=1$}{$\begin{cases}3y-2x=11\\3y=4x+7\end{cases}$}{$x=2,\ y=5$}
\QQRow{$\begin{cases}4x=3y-2\\4x-5y=-6\end{cases}$}{$x=1,\ y=2$}{$\begin{cases}2x+7y=5\\5y=3-2x\end{cases}$}{$x=-1,\ y=1$}
\QQRow{$\begin{cases}2x=5-y\\7y-2x=-13\end{cases}$}{$x=3,\ y=-1$}{$\begin{cases}5x+2y=12\\2y=3x-4\end{cases}$}{$x=2,\ y=1$}

\TypeQuestion{次の連立方程式を代入法で解きなさい。}{equated_expressions}
\QQRow{$\begin{cases}y=2x+1\\y=-x+7\end{cases}$}{$x=2,\ y=5$}{$\begin{cases}x=4y-5\\x=-2y+7\end{cases}$}{$x=3,\ y=2$}
\QQRow{$\begin{cases}y=-3x-2\\y=x+6\end{cases}$}{$x=-2,\ y=4$}{$\begin{cases}x=3-y\\x=2y+6\end{cases}$}{$x=4,\ y=-1$}
\QQRow{$\begin{cases}x=-3y+4\\x=y-8\end{cases}$}{$x=-5,\ y=3$}{$\begin{cases}y=5x-9\\y=3-x\end{cases}$}{$x=2,\ y=1$}
\QQRow{$\begin{cases}y=-2x+8\\y=3x+3\end{cases}$}{$x=1,\ y=6$}{$\begin{cases}x=2y+1\\x=13-y\end{cases}$}{$x=9,\ y=4$}
