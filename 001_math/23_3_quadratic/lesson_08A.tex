\MajorQuestionLesson{速さの利用問題}
\SetRowSpace{5mm}

\TypeQuestionSingle{}{douten1}
\FigProblemBlock{50mm}{fig/test_08_01}{\Text{直角三角形 ${\rm ABC}$ があり、$\Ang{B}=90^\circ$、$\Lseg{AB}=12$cm、$\Lseg{BC}=16$cmである。点 ${\rm P}$ は点 ${\rm A}$ を出発して辺 $\Lseg{AB}$ 上を点 ${\rm B}$ まで毎秒1cmの速さで動き、点 ${\rm Q}$ は点 ${\rm B}$ を出発して辺 $\Lseg{BC}$ 上を点 ${\rm C}$ まで毎秒2cmの速さで動く。2点 ${\rm P,Q}$ が同時に出発してから $t$ 秒後($0<t\leqq 8$)について、次の問いに答えなさい。}}
{
\QQ{$\Lseg{PB},\ \Lseg{BQ}$ の長さを $t$ を使って表しなさい。}{$\Lseg{PB}=12-t,\ \Lseg{BQ}=2t$}
\QQ{$\Tri{PBQ}$ の面積を $t$ を使って表しなさい。}{$12t-t^2$ cm$^2$}
\QQ{$\Tri{PBQ}$ の面積が32cm$^2$になるのは、出発してから何秒後か求めなさい。複数ある場合はすべて答えなさい。}{$4$秒後、$8$秒後}
}

\TypeQuestionSingle{}{douten2}
\FigProblemBlock{50mm}{fig/lesson_08_douten_02}{\Text{直角三角形 ${\rm ABC}$ があり、$\Ang{B}=90^\circ$、$\Lseg{AB}=12$cm、$\Lseg{BC}=16$cmである。点 ${\rm P}$ は点 ${\rm A}$ を出発して辺 $\Lseg{AB}$ 上を点 ${\rm B}$ まで毎秒2cmの速さで動き、点 ${\rm Q}$ は点 ${\rm C}$ を出発して辺 $\Lseg{BC}$ 上を点 ${\rm B}$ まで毎秒1cmの速さで動く。2点 ${\rm P,Q}$ が同時に出発してから $t$ 秒後($0<t\leqq 6$)について、次の問いに答えなさい。}}
{
\QQ{$\Lseg{PB},\ \Lseg{BQ}$ の長さを $t$ を使って表しなさい。}{$\Lseg{PB}=12-2t,\ \Lseg{BQ}=16-t$}
\QQ{$\Tri{PBQ}$ の面積を $t$ を使って表しなさい。}{$t^2-22t+96$ cm$^2$}
\QQ{$\Tri{PBQ}$ の面積が10cm$^2$になるのは、出発してから何秒後か求めなさい。複数ある場合はすべて答えなさい。}{$11-\sqrt{35}$秒後}
}

\QQ{$0<t\leqq 5$ のとき、$\Tri{DPQ}$ の面積を $t$ を使って表しなさい。}{$5t$ cm$^2$}

}
