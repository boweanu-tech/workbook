\TypeQuestionSingle{}{douten1}
\FigProblemBlock{50mm}{fig/test_08_01}{\Text{直角三角形 ${\rm ABC}$ があり、$\Ang{B}=90^\circ$、$\Lseg{AB}=12$cm、$\Lseg{BC}=16$cmである。点 ${\rm P}$ は点 ${\rm A}$ を出発して辺 $\Lseg{AB}$ 上を点 ${\rm B}$ まで毎秒1cmの速さで動き、点 ${\rm Q}$ は点 ${\rm B}$ を出発して辺 $\Lseg{BC}$ 上を点 ${\rm C}$ まで毎秒2cmの速さで動く。2点 ${\rm P,Q}$ が同時に出発してから $t$ 秒後($0<t\leqq 8$)について、次の問いに答えなさい。}}
{
\QQ{$\Lseg{PB},\ \Lseg{BQ}$ の長さを $t$ を使って表しなさい。}{$PB=12-t,\ BQ=2t$}
\QQ{$\Tri{PBQ}$ の面積を $t$ を使って表しなさい。}{$12t-t^2$ cm$^2$}
\QQ{$\Tri{PBQ}$ の面積が32cm$^2$になるのは、出発してから何秒後か求めなさい。}{$4$秒後、$8$秒後}
}

\TypeQuestionSingle{}{douten3}
\FigProblemBlock{50mm}{fig/lesson_08_douten_03}{\Text{1辺10cmの正方形 ${\rm ABCD}$ がある。点 ${\rm P}$ は点 ${\rm A}$ を出発し、辺 $\Lseg{AB}$、辺 $\Lseg{BC}$ の順に毎秒2cmの速さで動く。点 ${\rm Q}$ は点 ${\rm D}$ を出発して辺 $\Lseg{DC}$ 上を点 ${\rm C}$ まで毎秒1cmの速さで動く。2点 ${\rm P,Q}$ が同時に出発してから $t$ 秒後($0<t\leqq 10$)について、次の問いに答えなさい。}}
{
\QQ{$0<t\leqq 5$ のとき、$\Tri{DPQ}$ の面積を $t$ を使って表しなさい。}{$5t$ cm$^2$}
\QQ{$5<t\leqq 10$ のとき、$\Tri{DPQ}$ の面積を $t$ を使って表しなさい。}{$10t-t^2$ cm$^2$}
\QQ{$\Tri{DPQ}$ の面積が21cm$^2$になるのは、出発してから何秒後かすべて求めなさい。}{$\dfrac{21}{5}$秒後、$7$秒後}
}

\TypeQuestionSingle{}{douten4}
\FigProblemBlock{50mm}{fig/lesson_08_douten_04}{\Text{長方形 ${\rm ABCD}$ があり、$\Lseg{AB}=16$cm、$\Lseg{BC}=8$cmである。点 ${\rm P}$ は点 ${\rm A}$ を出発して辺 $\Lseg{AB}$ 上を点 ${\rm B}$ まで毎秒2cmの速さで動き、点 ${\rm B}$ に着くとすぐ折り返して、同じ速さで辺 $\Lseg{AB}$ 上を点 ${\rm A}$ に向かって動く。点 ${\rm Q}$ は点 ${\rm C}$ を出発して辺 $\Lseg{CD}$ 上を点 ${\rm D}$ まで毎秒1cmの速さで動く。2点 ${\rm P,Q}$ が同時に出発してから $t$ 秒後($0<t<16$)について、次の問いに答えなさい。}}
{
\QQ{$0<t\leqq 8$ のとき、台形 ${\rm APQD}$ の面積を $t$ を使って表しなさい。}{$64+4t$ cm$^2$}
\QQ{$8<t<16$ のとき、台形 ${\rm APQD}$ の面積を $t$ を使って表しなさい。}{$192-12t$ cm$^2$}
\QQ{台形 ${\rm APQD}$ の面積が80cm$^2$になるのは、出発してから何秒後かすべて求めなさい。}{$4$秒後、$\dfrac{28}{3}$秒後}
\QQ{台形 ${\rm APQD}$ の面積が台形 ${\rm BPQC}$ の面積のちょうど2倍になるのは、出発してから何秒後かすべて求めなさい。}{$\dfrac{16}{3}$秒後、$\dfrac{80}{9}$秒後}
}
