\TrigAnswerRow{$2\sqrt{3}\cos\theta-2\sin\theta$}{$4\sin\left(\theta+\dfrac{2\pi}{3}\right)$}{$\dfrac{1}{2}\sin\theta-\dfrac{\sqrt{3}}{2}\cos\theta$}{$\sin\left(\theta-\dfrac\pi3\right)$}

\TypeQuestion{次の式を $r\sin(\theta+\alpha)$ の形に表し、$r,\ \sin\alpha,\ \cos\alpha$ の値を求めなさい。ただし、$r>0$、$-\pi<\alpha\leqq\pi$ とする。}{synthesis_nonstandard_phase}
\TrigAnswerRow{$3\sin\theta-4\cos\theta$}{\AnswerThreeLines{$r=5$}{$\sin\alpha=-\dfrac45,\ \cos\alpha=\dfrac35$}{$5\sin(\theta+\alpha)$}}{$-2\sin\theta+3\cos\theta$}{\AnswerThreeLines{$r=\sqrt{13}$}{$\sin\alpha=\dfrac{3\sqrt{13}}{13},\ \cos\alpha=-\dfrac{2\sqrt{13}}{13}$}{$\sqrt{13}\sin(\theta+\alpha)$}}
\TrigAnswerRow{$-5\sin\theta-12\cos\theta$}{\AnswerThreeLines{$r=13$}{$\sin\alpha=-\dfrac{12}{13},\ \cos\alpha=-\dfrac5{13}$}{$13\sin(\theta+\alpha)$}}{$\dfrac{5}{2}\sin\theta+6\cos\theta$}{\AnswerThreeLines{$r=\dfrac{13}{2}$}{$\sin\alpha=\dfrac{12}{13},\ \cos\alpha=\dfrac5{13}$}{$\dfrac{13}{2}\sin(\theta+\alpha)$}}
\TrigAnswerRow{$\dfrac{1}{3}\sin\theta-\dfrac{2}{3}\cos\theta$}{\AnswerThreeLines{$r=\dfrac{\sqrt5}{3}$}{$\sin\alpha=-\dfrac{2\sqrt5}{5},\ \cos\alpha=\dfrac{\sqrt5}{5}$}{$\dfrac{\sqrt5}{3}\sin(\theta+\alpha)$}}{$4\cos\theta-7\sin\theta$}{\AnswerThreeLines{$r=\sqrt{65}$}{$\sin\alpha=\dfrac{4\sqrt{65}}{65},\ \cos\alpha=-\dfrac{7\sqrt{65}}{65}$}{$\sqrt{65}\sin(\theta+\alpha)$}}
