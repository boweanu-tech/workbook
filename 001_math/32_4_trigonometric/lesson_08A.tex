\MajorQuestionLesson{三角関数の合成と積和と和積の公式}
\SetRowSpace{4mm}

\TypeQuestion{次の式を $r\sin(\theta+\alpha)$ の形に表しなさい。ただし、$r>0$、$-\pi<\alpha\leqq\pi$ とする。}{synthesis_sine_form}
\QQRow{$\sin\theta+\cos\theta$}{}{$\sqrt{3}\sin\theta-\cos\theta$}{}
\QQRow{$-\sin\theta+\sqrt{3}\cos\theta$}{}{$\sin\theta-\cos\theta$}{}
\QQRow{$-2\sin\theta-2\cos\theta$}{}{$\cos\theta+\sqrt{3}\sin\theta$}{}
\QQRow{$2\sqrt{3}\cos\theta-2\sin\theta$}{}{$\dfrac{1}{2}\sin\theta-\dfrac{\sqrt{3}}{2}\cos\theta$}{}
\QQRow{$-\sqrt{3}\sin\theta-\cos\theta$}{}{$\dfrac{3}{2}\sin\theta+\dfrac{3}{2}\cos\theta$}{}

\TypeQuestion{次の式を $r\cos(\theta-\alpha)$ の形に表しなさい。ただし、$r>0$、$-\pi<\alpha\leqq\pi$ とする。}{synthesis_cosine_form}
\QQRow{$\sqrt{3}\cos\theta+\sin\theta$}{}{$\cos\theta-\sin\theta$}{}
\QQRow{$-\sin\theta-\sqrt{3}\cos\theta$}{}{$2\sin\theta-2\cos\theta$}{}
\QQRow{$3\sin\theta-\sqrt{3}\cos\theta$}{}{$-\dfrac{1}{2}\cos\theta-\dfrac{\sqrt{3}}{2}\sin\theta$}{}

\TypeQuestion{次の式を $r\sin(\theta+\alpha)$ の形に表し、$r,\ \sin\alpha,\ \cos\alpha$ の値を求めなさい。ただし、$r>0$、$-\pi<\alpha\leqq\pi$ とする。}{synthesis_nonstandard_phase}
\QQRow{$3\sin\theta-4\cos\theta$}{}{$-2\sin\theta+3\cos\theta$}{}
\QQRow{$-5\sin\theta-12\cos\theta$}{}{$\dfrac{5}{2}\sin\theta+6\cos\theta$}{}
\QQRow{$\dfrac{1}{3}\sin\theta-\dfrac{2}{3}\cos\theta$}{}{$4\cos\theta-7\sin\theta$}{}

\TypeQuestion{次の関数のグラフを $0\leqq\theta\leqq2\pi$ の範囲でかき、周期を求めなさい。}{synthesis_graph}
\QQRow{$y=\sin\theta+\cos\theta$}{}{$y=\sqrt{3}\cos\theta-\sin\theta$}{}
\QQRow{$y=2\sin\theta-2\cos\theta+1$}{}{$y=\sin2\theta+\sqrt{3}\cos2\theta$}{}

\LessonQuestionBreak

\TypeQuestion{次の積を、三角関数の和または差の形に直しなさい。}{product_sum_product_to_sum}
\QQRow{$2\sin3\theta\cos\theta$}{}{$\cos5\theta\cos2\theta$}{}
\QQRow{$\sin4\theta\sin\theta$}{}{$2\cos3\theta\sin2\theta$}{}
\QQRow{$\sin\left(\theta+\dfrac{\pi}{6}\right)\cos\left(\theta-\dfrac{\pi}{6}\right)$}{}{$\cos\left(\theta+\dfrac{\pi}{4}\right)\cos\left(\theta-\dfrac{\pi}{4}\right)$}{}

\TypeQuestion{次の和または差を、三角関数の積の形に直しなさい。}{product_sum_sum_to_product}
\QQRow{$\sin5\theta-\sin\theta$}{}{$\cos3\theta+\cos\theta$}{}
\QQRow{$\cos4\theta-\cos2\theta$}{}{$\sin2\theta+\sin\theta$}{}
\QQRow{$\sin\left(\theta+\dfrac{\pi}{3}\right)+\sin\left(\theta-\dfrac{\pi}{6}\right)$}{}{$\cos\left(\theta+\dfrac{\pi}{6}\right)-\cos\left(\theta-\dfrac{\pi}{3}\right)$}{}

\TypeQuestion{次の式の値を求めなさい。}{product_sum_calculate}
\QQRow{$\sin75^\circ\sin15^\circ$}{}{$\cos\dfrac{\pi}{12}\cos\dfrac{7\pi}{12}$}{}
\QQRow{$\sin105^\circ+\sin15^\circ$}{}{$\cos\dfrac{\pi}{12}-\cos\dfrac{5\pi}{12}$}{}
\QQRow{$\cos20^\circ\cos40^\circ\cos80^\circ$}{}{$2\sin10^\circ\sin50^\circ\sin70^\circ$}{}
