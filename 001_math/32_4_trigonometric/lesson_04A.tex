\MajorQuestionLesson{三角方程式}
\SetRowSpace{5mm}

\TypeQuestionSingle{次の方程式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{basic_sin}
\QQRowThree{$\sin\theta=\dfrac{1}{2}$}{}{$\sin\theta=-\dfrac{\sqrt{2}}{2}$}{}{$\sin\theta=1$}{}
\QQRowThree{$\sin\theta=-\dfrac{\sqrt{3}}{2}$}{}{$\sin\theta=0$}{}{$\sin\theta=\dfrac{\sqrt{3}}{2}$}{}

\TypeQuestionSingle{次の方程式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{basic_cos}
\QQRowThree{$\cos\theta=-\dfrac{1}{2}$}{}{$\cos\theta=\dfrac{\sqrt{2}}{2}$}{}{$\cos\theta=-1$}{}
\QQRowThree{$\cos\theta=\dfrac{\sqrt{3}}{2}$}{}{$\cos\theta=0$}{}{$\cos\theta=-\dfrac{\sqrt{3}}{2}$}{}

\TypeQuestionSingle{次の方程式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{basic_tan}
\QQRowThree{$\tan\theta=1$}{}{$\tan\theta=-\sqrt{3}$}{}{$\tan\theta=0$}{}
\QQRowThree{$\tan\theta=\dfrac{\sqrt{3}}{3}$}{}{$\tan\theta=-1$}{}{$\tan\theta=\sqrt{3}$}{}

\TypeQuestionSingle{次の方程式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{linear_form}
\QQRow{$2\sin\theta-\sqrt{3}=0$}{}{$\sqrt{2}\cos\theta+1=0$}{}
\QQRow{$3\tan\theta+\sqrt{3}=0$}{}{$4\cos\theta-2=0$}{}

\TypeQuestionSingle{次の方程式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{multiple_angle}
\QQRow{$\sin2\theta=\dfrac{\sqrt{3}}{2}$}{}{$\cos2\theta=-\dfrac{\sqrt{2}}{2}$}{}
\QQRow{$\tan2\theta=1$}{}{$\sin3\theta=-1$}{}
\QQRow{$\cos\left(\theta-\dfrac{\pi}{3}\right)=\dfrac{1}{2}$}{}{$\tan\left(\theta+\dfrac{\pi}{6}\right)=-\sqrt{3}$}{}

\TypeQuestionSingle{次の方程式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{quadratic_one_trig}
\QQRow{$2\sin^2\theta-3\sin\theta+1=0$}{}{$2\cos^2\theta+\cos\theta-1=0$}{}
\QQRow{$\tan^2\theta-\tan\theta=0$}{}{$4\sin^2\theta-1=0$}{}
\QQRow{$2\cos^2\theta-3\cos\theta=0$}{}{$\tan^2\theta-3=0$}{}

\TypeQuestionSingle{三角関数の相互関係を用いて、次の方程式を $0\leqq\theta<2\pi$ の範囲で解きなさい。}{identity_equation}
\QQRow{$2\sin^2\theta+3\cos\theta-3=0$}{}{$2\cos^2\theta+\sin\theta-1=0$}{}
\QQRow{$\sin^2\theta=3\cos^2\theta$}{}{$2\sin^2\theta-\cos^2\theta=\dfrac{1}{2}$}{}

\TypeQuestionDouble{$n$ を整数として、次の方程式の一般解を求めなさい。}{general_solution}
\QQRowThree{$\sin\theta=-\dfrac{1}{2}$}{}{$\cos\theta=\dfrac{\sqrt{2}}{2}$}{}{$\tan\theta=-\sqrt{3}$}{}
\QQRow{$\sin2\theta=\dfrac{\sqrt{3}}{2}$}{}{$\cos\left(\theta+\dfrac{\pi}{4}\right)=0$}{}
