% theme: trigonometric_inequalities
\MajorQuestionLesson{三角不等式}
\SetRowSpace{5mm}

\TypeQuestionSingle{次の不等式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{basic_inequality}
\QQRowThree{$\sin\theta>0$}{}{$\cos\theta\geqq-\dfrac{\sqrt{3}}{2}$}{}{$\tan\theta>1$}{}
\QQRowThree{$\tan\theta<0$}{}{$\sin\theta\leqq-\dfrac{\sqrt{2}}{2}$}{}{$\cos\theta>\dfrac{\sqrt{2}}{2}$}{}
\QQRowThree{$\cos\theta\leqq0$}{}{$\tan\theta\geqq-1$}{}{$\sin\theta\geqq\dfrac{\sqrt{3}}{2}$}{}
\QQRowThree{$\sin\theta>-\dfrac{\sqrt{3}}{2}$}{}{$\cos\theta<\dfrac{1}{2}$}{}{$\tan\theta\leqq-\sqrt{3}$}{}

\TypeQuestionSingle{次の不等式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{linear_form}
\QQRow{$2\sin\theta-1\geqq0$}{}{$\sqrt{2}\cos\theta+1<0$}{}
\QQRow{$\sqrt{3}\tan\theta-3\leqq0$}{}{$1-2\cos\theta>0$}{}
\QQRow{$2\sin\theta-3>0$}{}{$2\cos\theta-\sqrt{3}\geqq0$}{}
\QQRow{$\tan\theta+1>0$}{}{$2\cos\theta+3\geqq0$}{}

\TypeQuestionSingle{次の不等式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{multiple_angle}
\QQRow{$\sin2\theta>\dfrac{1}{2}$}{}{$\cos2\theta\leqq-\dfrac{\sqrt{2}}{2}$}{}
\QQRow{$\tan2\theta\geqq1$}{}{$\sin\left(\theta-\dfrac{\pi}{3}\right)<0$}{}
\QQRow{$\sin3\theta\leqq0$}{}{$\cos\left(\theta+\dfrac{\pi}{4}\right)>\dfrac{\sqrt{2}}{2}$}{}
\QQRow{$\tan\left(\theta-\dfrac{\pi}{6}\right)<0$}{}{$\cos3\theta\geqq\dfrac{1}{2}$}{}

\LessonPageBreak
\SetRowSpace{5mm}

\TypeQuestionSingle{次の不等式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{quadratic_one_trig}
\QQRow{$2\sin^2\theta-\sin\theta-1<0$}{}{$2\cos^2\theta+\cos\theta-1\geqq0$}{}
\QQRow{$4\sin^2\theta-3\leqq0$}{}{$\tan^2\theta-1>0$}{}
\QQRow{$2\cos^2\theta-3\cos\theta+1<0$}{}{$3\tan^2\theta-1\leqq0$}{}
\QQRow{$\sin^2\theta-\sin\theta\leqq0$}{}{$2\cos^2\theta+3\cos\theta+1>0$}{}
\QQRow{$\tan^2\theta-\sqrt{3}\tan\theta<0$}{}{$4\sin^2\theta+2\sin\theta-2>0$}{}

\TypeQuestionDouble{三角関数の相互関係を用いて、次の不等式を $0\leqq\theta<2\pi$ の範囲で解きなさい。}{identity_inequality}
\QQRow{$2\sin^2\theta+\cos\theta-2\geqq0$}{}{$2\cos^2\theta+3\sin\theta-2<0$}{}
\QQRow{$3\sin^2\theta-\cos^2\theta\geqq2$}{}{$\sin^2\theta-3\cos^2\theta<0$}{}

\TypeQuestionDouble{$n$ を整数として、次の不等式を満たす一般角 $\theta$ の範囲を求めなさい。}{general_solution}
\QQRow{$\sin\theta>\dfrac{1}{2}$}{}{$\cos\theta\leqq-\dfrac{\sqrt{2}}{2}$}{}
\QQRow{$\tan\theta\geqq\sqrt{3}$}{}{$\sin2\theta<0$}{}
\QQRow{$2\sin\theta-\sqrt{3}\leqq0$}{}{$\sqrt{2}\cos\theta+1>0$}{}
\QQRow{$\tan2\theta<0$}{}{$\sin\left(\theta-\dfrac{\pi}{6}\right)\geqq\dfrac{1}{2}$}{}
