\MajorQuestionLesson{公式の方程式・不等式への利用}
\SetRowSpace{4mm}

\TypeQuestion{次の方程式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{formula_application_equation}
\QQRow{$\cos2\theta=3\cos\theta-2$}{}{$\sin\theta-\cos\theta=-\sqrt{2}$}{}
\QQRow{$\cos2\theta=\sin\theta$}{}{$\sin\dfrac{\theta}{2}=\cos\theta$}{}
\QQRow{$\tan2\theta+\tan\theta=0$}{}{$\sin2\theta+\sin\theta=0$}{}
\QQRow{$\sin\theta+\sqrt{3}\cos\theta=\sqrt{3}$}{}{$\cos\theta+\cos\dfrac{\theta}{2}=0$}{}
\QQRow{$\sin3\theta=\sin\theta$}{}{$\sin2\theta-\sqrt{3}\cos2\theta=1$}{}
\QQRow{$\sqrt{3}\sin\theta-\cos\theta=1$}{}{$2\sin\theta+2\cos\theta=-\sqrt{2}$}{}
\QQRow{$\cos2\theta+\cos\theta=0$}{}{$\tan\dfrac{\theta}{2}=\sin\theta$}{}
\QQRow{$\sqrt{3}\cos\theta-\sin\theta=-1$}{}{$\cos3\theta=\cos\theta$}{}
\QQRow{$\sin\theta+\cos\theta=1$}{}{$\sqrt{3}\sin\theta+\cos\theta=-2$}{}
\QQRow{$\sin2\theta=\cos\theta$}{}{$\sin2\theta=2\sin\theta$}{}


\TypeQuestion{次の不等式を、$0\leqq\theta<2\pi$ の範囲で解きなさい。}{formula_application_inequality}
\QQRow{$\sin\theta+\cos\theta>1$}{}{$\sqrt{3}\sin\theta-\cos\theta\leqq0$}{}
\QQRow{$\cos\dfrac{\theta}{2}-\cos\theta\leqq0$}{}{$\sin\theta-\cos\theta\geqq-1$}{}
\QQRow{$\cos2\theta\leqq\cos\theta$}{}{$\cos2\theta+\sin\theta\geqq0$}{}
\QQRow{$\sin3\theta-\sin\theta<0$}{}{$\sin2\theta+\cos2\theta<1$}{}
\QQRow{$\sin2\theta>\sin\theta$}{}{$\sin^2\dfrac{\theta}{2}\leqq\cos\theta+\dfrac{1}{2}$}{}
\QQRow{$\sin\theta+\sqrt{3}\cos\theta\geqq\sqrt{3}$}{}{$\sqrt{3}\cos\theta-\sin\theta<-1$}{}
\QQ{$\cos3\theta+\cos\theta\geqq0$}{}

\LessonQuestionBreak

% 積和・和積の公式を履修しない場合は、ここからの2大問を省略する。
\TypeQuestion{積和・和積の公式を用いて、次の方程式を $0\leqq\theta<2\pi$ の範囲で解きなさい。}{product_sum_equation}
\QQRow{$\sin3\theta+\sin\theta=0$}{}{$\cos2\theta-\cos\theta=0$}{}
\QQRow{$2\sin2\theta\cos\theta=\sin3\theta$}{}{$2\cos2\theta\cos\theta=\cos3\theta+\dfrac{1}{2}$}{}

\TypeQuestion{積和・和積の公式を用いて、次の不等式を $0\leqq\theta<2\pi$ の範囲で解きなさい。}{product_sum_inequality}
\QQRow{$\sin3\theta+\sin\theta>0$}{}{$\cos3\theta-\cos\theta\leqq0$}{}
\QQRow{$2\sin2\theta\cos\theta\leqq\sin3\theta+\dfrac{1}{2}$}{}{$2\cos2\theta\cos\theta>\cos3\theta+\dfrac{\sqrt{3}}{2}$}{}
