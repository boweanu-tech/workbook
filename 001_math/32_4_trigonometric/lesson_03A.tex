% (2) SVG：lesson_03_coordinates_tan_01.svg
%     theta範囲 -pi/2～pi/2、縦罫線 pi/4 間隔。ラベルは -pi/2, -pi/4, O, pi/4, pi/2。

% (3) SVG：lesson_03_coordinates_cos_01.svg
%     theta範囲 -pi～pi、縦罫線 pi/2 間隔。ラベルは -pi, -pi/2, O, pi/2, pi。
%     y=cos theta を描き、P(-pi,-1), Q(0,1), R(pi/2,0) を黒点で示す。
%     座標の投影補助線は入れず、点名は曲線・軸と重ならない位置に置く。
% (4) SVG：lesson_03_coordinates_sin_02.svg
%     theta範囲 0～2pi、縦罫線 pi/6 間隔。ラベルは O, pi/2, pi, 3pi/2, 2pi。
%     y表示範囲は -2.4～2.4、y=+-1, +-2 に目盛を置く。
%     y=2sin theta を描き、P(pi/6,1), Q(pi/2,2), R(5pi/6,1) を黒点で示す。
%     座標の投影補助線は入れず、点名は曲線・軸と重ならない位置に置く。
\QQRow
  {$y=\sin\theta\quad(0\leqq\theta\leqq2\pi)$}{}
  {$y=\tan\theta\quad\left(-\dfrac{\pi}{2}<\theta<\dfrac{\pi}{2}\right)$}{}
\QQRow
  {$y=\cos\theta\quad(-\pi\leqq\theta\leqq\pi)$}{}
  {$y=2\sin\theta\quad(0\leqq\theta\leqq2\pi)$}{}

\TypeQuestionSingle{次の関数のグラフをかき、その周期を求めなさい。}{horizontal_transform}
% (1) SVG：lesson_03_scale_sin_01_q.svg / lesson_03_scale_sin_01_a.svg
%     theta範囲 -pi～pi、縦罫線 pi/4 間隔。ラベルは -pi, -pi/2, O, pi/2, pi。
\QQ{$y=\sin2\theta\quad(-\pi\leqq\theta\leqq\pi)$}{}
% (2) SVG：lesson_03_scale_sin_02_q.svg / lesson_03_scale_sin_02_a.svg
%     theta範囲 -3pi～3pi、縦罫線 3pi/2 間隔。全罫線に -3pi, -3pi/2, O, 3pi/2, 3pi を表示。
\QQ{$y=\sin\dfrac{\theta}{3}\quad(-3\pi\leqq\theta\leqq3\pi)$}{}
% (3) SVG：lesson_03_scale_cos_01_q.svg / lesson_03_scale_cos_01_a.svg
%     theta範囲 -2pi～2pi、縦罫線 pi 間隔。全罫線に -2pi, -pi, O, pi, 2pi を表示。
\QQ{$y=\cos\dfrac{\theta}{2}\quad(-2\pi\leqq\theta\leqq2\pi)$}{}
% (4) SVG：lesson_03_scale_cos_02_q.svg / lesson_03_scale_cos_02_a.svg
%     theta範囲 -2pi/3～2pi/3、縦罫線 pi/6 間隔。ラベルは -2pi/3, -pi/3, O, pi/3, 2pi/3。
%     +-pi/6 は罫線・小目盛のみで、ラベルは省略する。
\QQ{$y=\cos3\theta\quad\left(-\dfrac{2\pi}{3}\leqq\theta\leqq\dfrac{2\pi}{3}\right)$}{}
% (5) SVG：lesson_03_scale_tan_01_q.svg / lesson_03_scale_tan_01_a.svg
%     theta範囲 -pi/2～pi/2、縦罫線 pi/8 間隔。ラベルは -pi/2, -pi/4, O, pi/4, pi/2。
%     解答図では theta=+-pi/4 を漸近線とし、theta=+-pi/2 の有限な開端点 (theta,0) は白丸。
\QQ{$y=\tan2\theta\quad\left(-\dfrac{\pi}{2}<\theta<\dfrac{\pi}{2}\right)$}{}

\TypeQuestionSingle{次の関数のグラフをかき、その周期を求めなさい。}{translation}
% (1) SVG：lesson_03_shift_sin_01_q.svg / lesson_03_shift_sin_01_a.svg
%     theta範囲 -pi～2pi、縦罫線 pi/6 間隔。ラベルは -pi, -pi/2, O, pi/2, pi, 3pi/2, 2pi。
%     pi/6 単位の罫線は無表示の小目盛とし、位相 pi/3 の位置を正確に取れるようにする。
\QQ{$y=\sin\left(\theta-\dfrac{\pi}{3}\right)\quad(-\pi\leqq\theta\leqq2\pi)$}{}
% (2) SVG：lesson_03_shift_sin_02_q.svg / lesson_03_shift_sin_02_a.svg
%     theta範囲 -5pi/2～3pi/2、縦罫線 pi/2 間隔。
%     ラベルは -5pi/2, -3pi/2, -pi/2, pi/2, 3pi/2。theta=0 は O のみ表示。
\QQ{$y=\sin\dfrac{1}{2}\left(\theta+\dfrac{\pi}{2}\right)\quad\left(-\dfrac{5\pi}{2}\leqq\theta\leqq\dfrac{3\pi}{2}\right)$}{}
% (3) SVG：lesson_03_shift_cos_01_q.svg / lesson_03_shift_cos_01_a.svg
%     theta範囲 -pi～pi、縦罫線 pi/12 間隔。ラベルは -pi, -pi/2, -pi/6, O, pi/2, pi。
%     pi/12 単位の罫線は薄くし、位相 -pi/6 と零点 pi/12+k*pi/2 を正確に取れるようにする。
\QQ{$y=\cos2\left(\theta+\dfrac{\pi}{6}\right)\quad(-\pi\leqq\theta\leqq\pi)$}{}
% (4) SVG：lesson_03_shift_cos_02_q.svg / lesson_03_shift_cos_02_a.svg
%     theta範囲 -pi～2pi、縦罫線 pi/4 間隔。ラベルは -pi, -pi/2, O, pi/2, pi, 3pi/2, 2pi。
\QQ{$y=\cos\left(\theta-\dfrac{\pi}{4}\right)\quad(-\pi\leqq\theta\leqq2\pi)$}{}
% (5) SVG：lesson_03_shift_tan_01_q.svg / lesson_03_shift_tan_01_a.svg
%     theta範囲 -3pi/8～5pi/8、縦罫線 pi/8 間隔。
%     ラベルは -3pi/8, -pi/8, O, pi/8, 3pi/8, 5pi/8。
%     解答図では theta=-3pi/8, pi/8, 5pi/8 を漸近線とする。
\QQ{$y=\tan2\left(\theta+\dfrac{\pi}{8}\right)\quad\left(-\dfrac{3\pi}{8}<\theta<\dfrac{5\pi}{8}\right)$}{}

  {$y=\cos\left(\theta+\dfrac{\pi}{3}\right)\quad(0\leqq\theta\leqq\pi)$}{}
\QQRow
  {$y=2\sin\theta\quad\left(-\dfrac{\pi}{6}\leqq\theta\leqq\dfrac{\pi}{2}\right)$}{}
  {$y=-3\cos\theta+1\quad(0\leqq\theta\leqq\pi)$}{}
\QQRow
  {$y=\sin2\theta\quad\left(0\leqq\theta\leqq\dfrac{2\pi}{3}\right)$}{}
