\MajorQuestionLesson{三角関数のグラフと最大・最小}

\SetRowSpace{2mm}

\TypeQuestionDouble{次の関数のグラフ上に示された点 ${\rm P},\ {\rm Q},\ {\rm R}$ の座標をそれぞれ求めなさい。}{graph_coordinates}
% 図生成工程：各小問に、指定範囲の基本グラフと3点 P, Q, R を示す。
% 各点は、目盛や補助線から座標を直接読むのではなく、基本グラフの特徴から座標を判断する位置に置く。
% (1) P,Q,R の theta 座標：pi/2, pi, 3pi/2
% (2) P,Q,R の theta 座標：-pi/4, 0, pi/4
\QQRow
  {$y=\sin\theta\quad(0\leqq\theta\leqq2\pi)$}{}
  {$y=\tan\theta\quad\left(-\dfrac{\pi}{2}<\theta<\dfrac{\pi}{2}\right)$}{}

% 図生成工程：各小問に、横軸を弧度法で目盛った座標平面を挿入する。
%\thetaの係数が分数・整数かで揃っているので、ランダムに。
%ただ、小問がsin 4つ→cos 4つ→tan 4つ に並ぶようにして。

  {$y=\tan3\theta\quad\left(-\dfrac{\pi}{3}\leqq\theta\leqq\dfrac{\pi}{3}\right)$}{}

\QQRow
  {$y=\cos3\theta\quad\left(-\dfrac{2\pi}{3}\leqq\theta\leqq\dfrac{2\pi}{3}\right)$}{}
  {$y=\tan\dfrac{\theta}{2}\quad(-\pi<\theta<\pi)$}{}
\QQRow
  {$y=\sin3\theta\quad\left(-\dfrac{2\pi}{3}\leqq\theta\leqq\dfrac{2\pi}{3}\right)$}{}
  {$y=\cos\dfrac{\theta}{3}\quad(-3\pi\leqq\theta\leqq3\pi)$}{}
\QQRow
  {$y=\tan2\theta\quad\left(-\dfrac{\pi}{2}<\theta<\dfrac{\pi}{2}\right)$}{}
  {$y=\sin\dfrac{\theta}{2}\quad(-2\pi\leqq\theta\leqq2\pi)$}{}
\QQRow
  {$y=\cos2\theta\quad(-\pi\leqq\theta\leqq\pi)$}{}
  {$y=\tan\dfrac{\theta}{3}\quad\left(-\dfrac{3\pi}{2}<\theta<\dfrac{3\pi}{2}\right)$}{}

\TypeQuestionDouble{次の関数の最大値と最小値を、それぞれそのときの $\theta$ の値とともに求めなさい。}{basic_extrema}
\QQRow
  {$y=\sin\left(\theta-\dfrac{\pi}{6}\right)\quad(0\leqq\theta\leqq\pi)$}{}
  {$y=\cos\left(\theta+\dfrac{\pi}{3}\right)\quad(0\leqq\theta\leqq\pi)$}{}
\QQRow
  {$y=2\sin\theta\quad\left(-\dfrac{\pi}{6}\leqq\theta\leqq\dfrac{\pi}{2}\right)$}{}
  {$y=-3\cos\theta+1\quad(0\leqq\theta\leqq\pi)$}{}
\QQRow
  {$y=\sin2\theta\quad\left(0\leqq\theta\leqq\dfrac{2\pi}{3}\right)$}{}
  {$y=\cos2\theta\quad\left(-\dfrac{\pi}{4}\leqq\theta\leqq\dfrac{\pi}{3}\right)$}{}
\QQ{$y=2\cos\left(\theta-\dfrac{\pi}{4}\right)-1\quad(0\leqq\theta\leqq\pi)$}{}
\QQ{$y=-\sin\left(\theta+\dfrac{\pi}{6}\right)+2\quad\left(-\dfrac{\pi}{3}\leqq\theta\leqq\dfrac{\pi}{2}\right)$}{}

\TypeQuestionDouble{$-1\leqq\sin\theta\leqq1$ または $-1\leqq\cos\theta\leqq1$ を利用して、次の関数の最大値と最小値を求めなさい。}{quadratic_one_trig}
\QQRow
  {$y=2\sin^2\theta-4\sin\theta+1$}{}
  {$y=-3\cos^2\theta+3\cos\theta+2$}{}
\QQ{$y=\sin^2\theta+\sin\theta-1$}{}%\thetaの変域アリにして
\QQ{$y=2\cos^2\theta+\cos\theta-3$}{}%\thetaの変域アリにして

\TypeQuestionDouble{三角関数の相互関係を利用して1種類の三角関数で表し、次の関数の最大値と最小値を求めなさい。}{quadratic_with_identity}
\QQRow
  {$y=2\sin^2\theta+3\cos\theta$}{}
  {$y=3\cos^2\theta-2\sin\theta$}{}
\QQ{$y=\sin^2\theta+2\cos\theta+2$}{}%\thetaの変域アリにして
\QQ{$y=2\cos^2\theta+4\sin\theta-1$}{}%\thetaの変域アリにして
