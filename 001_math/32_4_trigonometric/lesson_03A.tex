\TypeQuestionSingle{次の関数の最大値と最小値を、それぞれそのときの $\theta$ の値とともに求めなさい。}{basic_extrema}
\ExtremaQQRow
  {$y=\sin\left(\theta-\dfrac{\pi}{6}\right)\quad(0\leqq\theta\leqq\pi)$}
  {\AnswerTwoLines{最大値 $1$($\theta=\dfrac{2\pi}{3}$)}{最小値 $-\dfrac{1}{2}$($\theta=0$)}}
  {$y=\cos\left(\theta+\dfrac{\pi}{3}\right)\quad(0\leqq\theta\leqq\pi)$}
  {\AnswerTwoLines{最大値 $\dfrac{1}{2}$($\theta=0$)}{最小値 $-1$($\theta=\dfrac{2\pi}{3}$)}}
\ExtremaQQRow
  {$y=2\sin\theta\quad\left(-\dfrac{\pi}{6}\leqq\theta\leqq\dfrac{\pi}{2}\right)$}
  {\AnswerTwoLines{最大値 $2$($\theta=\dfrac{\pi}{2}$)}{最小値 $-1$($\theta=-\dfrac{\pi}{6}$)}}
  {$y=-3\cos\theta+1\quad(0\leqq\theta\leqq\pi)$}
  {\AnswerTwoLines{最大値 $4$($\theta=\pi$)}{最小値 $-2$($\theta=0$)}}
\ExtremaQQRow
  {$y=\sin2\theta\quad\left(0\leqq\theta\leqq\dfrac{2\pi}{3}\right)$}
  {\AnswerTwoLines{最大値 $1$($\theta=\dfrac{\pi}{4}$)}{最小値 $-\dfrac{\sqrt{3}}{2}$($\theta=\dfrac{2\pi}{3}$)}}
  {$y=\cos2\theta\quad\left(-\dfrac{\pi}{4}\leqq\theta\leqq\dfrac{\pi}{3}\right)$}
  {\AnswerTwoLines{最大値 $1$($\theta=0$)}{最小値 $-\dfrac{1}{2}$($\theta=\dfrac{\pi}{3}$)}}
\QQ{$y=2\cos\left(\theta-\dfrac{\pi}{4}\right)-1\quad(0\leqq\theta\leqq\pi)$}
  {\AnswerTwoLines{最大値 $1$($\theta=\dfrac{\pi}{4}$)}{最小値 $-1-\sqrt{2}$($\theta=\pi$)}}
\QQ{$y=-\sin\left(\theta+\dfrac{\pi}{6}\right)+2\quad\left(-\dfrac{\pi}{3}\leqq\theta\leqq\dfrac{\pi}{2}\right)$}
  {\AnswerTwoLines{最大値 $\dfrac{5}{2}$($\theta=-\dfrac{\pi}{3}$)}{最小値 $1$($\theta=\dfrac{\pi}{3}$)}}

\AnswerColumnBreak
\TypeQuestionSingle{次の関数の最大値と最小値を、それぞれそのときの $\theta$ の値とともに求めなさい。}{quadratic_one_trig}
\ExtremaQQRow
  {$y=2\sin^2\theta-4\sin\theta+1$}
  {\AnswerThreeLines{最大値 $7$($\theta=\dfrac{3\pi}{2}+2n\pi$)}{最小値 $-1$($\theta=\dfrac{\pi}{2}+2n\pi$)}{($n$ は整数)}}
  {$y=-3\cos^2\theta+3\cos\theta+2$}
  {\AnswerThreeLines{最大値 $\dfrac{11}{4}$($\theta=2n\pi\pm\dfrac{\pi}{3}$)}{最小値 $-4$($\theta=(2n+1)\pi$)}{($n$ は整数)}}
\QQ{$y=\sin^2\theta+\sin\theta-1\quad\left(0\leqq\theta\leqq\dfrac{\pi}{6}\right)$}
  {\AnswerTwoLines{最大値 $-\dfrac{1}{4}$($\theta=\dfrac{\pi}{6}$)}{最小値 $-1$($\theta=0$)}}
\QQ{$y=2\cos^2\theta+\cos\theta-3\quad\left(0\leqq\theta\leqq\dfrac{2\pi}{3}\right)$}
  {\AnswerTwoLines{最大値 $0$($\theta=0$)}{最小値 $-\dfrac{25}{8}$($\cos\theta=-\dfrac{1}{4}$)}}


\TypeQuestionSingle{次の関数の最大値と最小値を、それぞれそのときの $\theta$ の値とともに求めなさい。}{quadratic_with_identity}
