\TypeQuestion{$x=2$、$y=-3$ のとき、次の式を簡単にしてから値を求めなさい。}{expression_substitution_distribution_integer}
\QQRow{$(x+2y)-3(2x+y)$}{}{$4(x-y)-2(3x+2y)$}{}
\QQRow{$-2(3x-y)+(5x+4y)$}{}{$3(x+2y)-(2x-5y)$}{}
\QQRow{$(7x-3y)-2(2x+y)$}{}{$-4(x-y)+3(2x-3y)$}{}

\TypeQuestion{$x=\dfrac32$、$y=-\dfrac23$ のとき、次の式を簡単にしてから値を求めなさい。}{expression_substitution_distribution_fraction}
\QQRow{$\dfrac12(x+2y)-\dfrac13(3x-y)$}{}{$\dfrac23(2x-y)-\dfrac12(x+3y)$}{}
\QQRow{$\dfrac34(2x+y)-\dfrac12(x-2y)$}{}{$-\dfrac13(3x-2y)+\dfrac56(x+y)$}{}
\QQRow{$\dfrac52(x-y)-\dfrac34(2x+y)$}{}{$\dfrac16(4x-3y)-\dfrac23(x+2y)$}{}

\TypeQuestion{$x=-2$、$y=3$ のとき、次の式を簡単にしてから値を求めなさい。}{expression_substitution_monomial_integer}
\QQRow{$(3xy)(-2x^2)$}{}{$12x^3y\div(3xy)$}{}
\QQRow{$(-4x^2y)(2xy^2)$}{}{$-18x^2y^3\div(-6xy^2)$}{}
\QQRow{$(5x^2)(-3xy)\div(5x)$}{}{$24x^3y^2\div(-4xy)\times y$}{}

\TypeQuestion{$x=\dfrac23$、$y=-\dfrac32$ のとき、次の式を簡単にしてから値を求めなさい。}{expression_substitution_monomial_fraction}
\QQRow{$\left(\dfrac34xy\right)(2x^2)$}{}{$\dfrac32x^2y\div(2xy)$}{}
\QQRow{$\left(-\dfrac56x^2\right)(3xy)$}{}{$\dfrac49x^3y^2\div\left(\dfrac23xy\right)$}{}
\QQRow{$\dfrac54xy^2\div\left(-\dfrac32y\right)\times x$}{}{$\left(-\dfrac23x^2y\right)\left(\dfrac38y\right)\div x$}{}
