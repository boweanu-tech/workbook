\MajorQuestionLesson{一般動詞の疑問文と短い答え}

\TypeQuestion{英文を書き写し、日本語に訳しなさい。}{generalverb_question_copy_translate}
\EngCopyTranslateSplit{Do you play tennis?}{}
\EngCopyTranslate{Do they study English every day?}{}

\TypeQuestion{（　）内の適切な語を丸で囲み、完成した正しい英文を書きなさい。}{generalverb_question_choose}
\EngWriteQuestion{\EngSentence{\EngChoice{Do / Are} you study English?}}{}
\EngWriteQuestion{\EngSentence{\EngChoice{Do / Are} they play soccer?}}{}

\LessonPageBreak

\TypeQuestion{選択肢から適切な語を選んで空欄を補い、完成した正しい英文を書きなさい。}{generalverb_question_fill_choice}
\EngWriteQuestion{\EngSentence{\EngBlank[12mm] you like music?}\par\EngChoice{Do / Are / Is}}{}
\EngWriteQuestion{\EngSentence{\EngBlank[12mm] they use this room?}\par\EngChoice{Do / Are / Am}}{}

\TypeQuestion{疑問文に対する答えとして最も適切なものを一つ選び、その記号を丸で囲みなさい。}{generalverb_question_response_choice}
\QQTall{%
  \EngSentence{Do you play baseball?}\par\vspace{2mm}%
  （ア）\EngText{Yes, I am.}\qquad
  （イ）\EngText{Yes, I do.}\qquad
  （ウ）\EngText{Yes, you do.}%
}{}
\QQTall{%
  \EngSentence{Do they study math?}\par\vspace{2mm}%
  （ア）\EngText{No, they don't.}\qquad
  （イ）\EngText{No, they aren't.}\qquad
  （ウ）\EngText{No, I don't.}%
}{}

\TypeQuestion{指示に従って英文を書き換えなさい。}{generalverb_question_transform}
\EngWriteQuestion{\EngSentence{You play tennis.}\quad〔疑問文に〕}{}
\EngWriteQuestion{\EngSentence{They study English every day.}\quad〔疑問文に〕}{}

\LessonPageBreak

\TypeQuestion{日本語に合うように、〔　〕内のすべての語句を並べ替えて、正しい英文を作りなさい。}{generalverb_question_reorder}
\EngWriteQuestion{あなたはテニスをしますか。\par\EngReorder{you / tennis / Do / play}}{}
\EngWriteQuestion{彼らはこのコンピューターを使いますか。\par\EngReorder{this computer / they / use / Do}}{}

\TypeQuestion{英文の誤りを直し、正しい英文を書きなさい。}{generalverb_question_error_correct}
\EngWriteQuestion{\EngSentence{You do play soccer?}}{}
\EngWriteQuestion{\EngSentence{Do you are like music?}}{}
