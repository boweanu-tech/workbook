\MajorQuestionLesson{一般動詞の疑問文と短い答え}

\TypeQuestion{英文を書き写し、日本語に訳しなさい。}{generalverb_question_copy_translate}
\EngCopyTranslateSplit{Do you play tennis?}{}
\EngCopyTranslateSplit{Do they study English every day?}{}
\EngCopyTranslateSplit{Do you like music?}{}
\EngCopyTranslateSplit{Do they use this room after school?}{}
\EngCopyTranslateSplit{Are you a student?}{}
\EngCopyTranslateSplit{Are they busy?}{}

\LessonPageBreak

\TypeQuestion{日本語に合うように、（　）内では適切な語を丸で囲み、空欄には選択肢から適切な語を入れなさい。また、完成した正しい英文を書きなさい。}{generalverb_question_fill_choice}
\EngWriteQuestion{あなたは英語を勉強しますか。\par\EngSentence{\EngChoice{Do / Are} you study English?}}{}
\EngWriteQuestion{彼らはサッカーをしますか。\par\EngSentence{\EngChoice{Do / Are} they play soccer?}}{}
\EngWriteQuestion{あなたは生徒ですか。\par\EngSentence{\EngChoice{Do / Are} you a student?}}{}
\EngWriteQuestion{私は犬が好きではありません。\par\EngSentence{I \EngChoice{don't / am not} like dogs.}}{}
\EngWriteQuestion{彼らは忙しくありません。\par\EngSentence{They \EngBlank[18mm] busy.}\par\EngChoice{aren't / don't / not}}{}
\EngWriteQuestion{私たちはテニスをします。\par\EngSentence{We \EngBlank[18mm] tennis.}\par\EngChoice{play / are / do}}{}

\LessonPageBreak

\TypeQuestion{疑問文に対する答えとして最も適切なものを一つ選び、その記号を丸で囲みなさい。}{generalverb_question_response_choice}
\QQTall{\EngSentence{Do you play baseball?}\par\vspace{2mm}（ア）\EngText{Yes, I am.}\qquad（イ）\EngText{Yes, I do.}\qquad（ウ）\EngText{Yes, you do.}}{}
\QQTall{\EngSentence{Do they study math?}\par\vspace{2mm}（ア）\EngText{No, they don't.}\qquad（イ）\EngText{No, they aren't.}\qquad（ウ）\EngText{No, I don't.}}{}
\QQTall{\EngSentence{Are you a student?}\par\vspace{2mm}（ア）\EngText{Yes, I do.}\qquad（イ）\EngText{Yes, you are.}\qquad（ウ）\EngText{Yes, I am.}}{}
\QQTall{\EngSentence{Are they busy?}\par\vspace{2mm}（ア）\EngText{No, they aren't.}\qquad（イ）\EngText{No, they don't.}\qquad（ウ）\EngText{No, we aren't.}}{}
\QQTall{\EngSentence{Is this your book?}\par\vspace{2mm}（ア）\EngText{Yes, this is.}\qquad（イ）\EngText{Yes, it is.}\qquad（ウ）\EngText{Yes, I do.}}{}
\QQTall{\EngSentence{Is he your friend?}\par\vspace{2mm}（ア）\EngText{Yes, he is.}\qquad（イ）\EngText{Yes, he does.}\qquad（ウ）\EngText{Yes, you are.}}{}

\LessonPageBreak

\TypeQuestion{もとの英文を疑問文に書き換え、〔　〕内の指示に従ってその疑問文に短く答えなさい。}{generalverb_question_transform}
\EngWriteQuestion{\EngSentence{You play tennis.}\quad〔\EngText{Yes}で〕}{}
\EngWriteQuestion{\EngSentence{They study English every day.}\quad〔\EngText{No}で〕}{}
\EngWriteQuestion{\EngSentence{You are busy.}\quad〔\EngText{Yes}で〕}{}
\EngWriteQuestion{\EngSentence{They are students.}\quad〔\EngText{No}で〕}{}
\EngWriteQuestion{\EngSentence{This is your book.}\quad〔\EngText{Yes}で〕}{}
\EngWriteQuestion{\EngSentence{You like dogs.}\quad〔\EngText{No}で〕}{}

\LessonPageBreak

\TypeQuestion{日本語に合うように、〔　〕内のすべての語句を並べ替えて、正しい英文を作りなさい。}{generalverb_question_reorder}
\EngWriteQuestion{あなたはテニスをしますか。\par\EngReorder{you / tennis / do / play}}{}
\EngWriteQuestion{彼らはこのコンピューターを使いますか。\par\EngReorder{this computer / they / use / do}}{}
\EngWriteQuestion{あなたは生徒ですか。\par\EngReorder{you / a student / are}}{}
\EngWriteQuestion{彼らは忙しいですか。\par\EngReorder{busy / they / are}}{}
\EngWriteQuestion{これはあなたの本ですか。\par\EngReorder{your book / this / is}}{}
\EngWriteQuestion{あなたは音楽が好きですか。\par\EngReorder{music / you / do / like}}{}
