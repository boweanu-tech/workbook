\MajorQuestionLesson{一般動詞の否定文}

\TypeQuestion{英文を書き写し、日本語に訳しなさい。}{generalverb_negative_copy_translate}
\EngCopyTranslate{I do not play soccer.}{}
\EngCopyTranslate{We don't study math on Sundays.}{}

\TypeQuestion{意味が同じになるように、短縮形を使った英文を書きなさい。}{generalverb_negative_contraction}
\EngWriteQuestion{\EngSentence{I do not use this computer.}}{}
\EngWriteQuestion{\EngSentence{They do not like cats.}}{}

\LessonPageBreak

\TypeQuestion{（　）内の適切な語を丸で囲み、完成した正しい英文を書きなさい。}{generalverb_negative_choose}
\EngWriteQuestion{\EngSentence{I \EngChoice{do not / am not} play baseball.}}{}
\EngWriteQuestion{\EngSentence{We \EngChoice{don't / aren't} read this book.}}{}

\TypeQuestion{選択肢から適切な語を選んで空欄を補い、完成した正しい英文を書きなさい。}{generalverb_negative_fill_choice}
\EngWriteQuestion{\EngSentence{You \EngBlank[22mm] tennis after school.}\par\EngChoice{do not play / are not / not play}}{}
\EngWriteQuestion{\EngSentence{They \EngBlank[22mm] English on Sundays.}\par\EngChoice{don't study / aren't study / not study}}{}

\TypeQuestion{指示に従って英文を書き換えなさい。}{generalverb_negative_transform}
\EngWriteQuestion{\EngSentence{I like dogs.}\quad〔否定文に〕}{}
\EngWriteQuestion{\EngSentence{We use this room.}\quad〔否定文に〕}{}
\EngWriteQuestion{\EngSentence{They play baseball.}\quad〔否定文に〕}{}

\LessonPageBreak

\TypeQuestion{日本語に合うように、〔　〕内のすべての語句を並べ替えて、正しい英文を作りなさい。}{generalverb_negative_reorder}
\EngWriteQuestion{私はテニスをしません。\par\EngReorder{tennis / do not / I / play}}{}
\EngWriteQuestion{私たちは放課後に英語を勉強しません。\par\EngReorder{after school / don't / English / We / study}}{}

\TypeQuestion{英文の誤りを直し、正しい英文を書きなさい。}{generalverb_negative_error_correct}
\EngWriteQuestion{\EngSentence{I not play soccer.}}{}
\EngWriteQuestion{\EngSentence{They don't are study math.}}{}
