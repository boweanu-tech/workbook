\MajorQuestionLesson{do・doesの使い分け}
\TypeQuestion{英文や会話文を書き写し、日本語に訳しなさい。}{generalverb_do_does_copy_translate}
\SetRowSpace{4mm}
\EngCopyTranslate{Do your brothers play tennis?\enspace---\enspace Yes, they do.}{}
\EngCopyTranslate{Does your brother play tennis?\enspace---\enspace No, he doesn't.}{}
\EngCopyTranslateSplit{Ken and Yui don't like music.}{}
\EngCopyTranslateSplit{Ken doesn't like music.}{}
\EngCopyTranslate{Is your sister busy?\enspace---\enspace Yes, she is.}{}
