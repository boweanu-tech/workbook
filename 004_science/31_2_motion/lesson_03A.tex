\MajorQuestionLesson{位置・変位・速さ・速度}
\SetRowSpace{4mm}

\TypeQuestionSingle{右向きを正として、数直線上の位置 $x$ を答えなさい。}{position}
\QQRow{原点から右へ $3.0\,\mathrm{m}$ の点}{$x=+3.0\,\mathrm{m}$}{原点から左へ $2.5\,\mathrm{m}$ の点}{$x=-2.5\,\mathrm{m}$}
\QQRow{原点から右へ $12\,\mathrm{m}$ の点}{$x=+12\,\mathrm{m}$}{原点から左へ $0.80\,\mathrm{m}$ の点}{$x=-0.80\,\mathrm{m}$}

\TypeQuestionSingle{変位 $\Delta x=x_2-x_1$ を求めなさい。}{displacement}
\QQRow{$x_1=2.0\,\mathrm{m}$ から $x_2=7.0\,\mathrm{m}$}{$\Delta x=+5.0\,\mathrm{m}$}{$x_1=6.0\,\mathrm{m}$ から $x_2=1.0\,\mathrm{m}$}{$\Delta x=-5.0\,\mathrm{m}$}
\QQRow{$x_1=-3.0\,\mathrm{m}$ から $x_2=4.0\,\mathrm{m}$}{$\Delta x=+7.0\,\mathrm{m}$}{$x_1=2.5\,\mathrm{m}$ から $x_2=-1.0\,\mathrm{m}$}{$\Delta x=-3.5\,\mathrm{m}$}
\QQRow{$x_1=-4.0\,\mathrm{m}$ から $x_2=-9.0\,\mathrm{m}$}{$\Delta x=-5.0\,\mathrm{m}$}{$x_1=-6.0\,\mathrm{m}$ から $x_2=-1.5\,\mathrm{m}$}{$\Delta x=+4.5\,\mathrm{m}$}

\TypeQuestionSingle{次の運動の移動距離と変位を求めなさい。}{distance_displacement}
\QQ{$x=1.0\,\mathrm{m}$ から $x=6.0\,\mathrm{m}$ まで直線上を進んだ。}{移動距離 $5.0\,\mathrm{m}$、変位 $+5.0\,\mathrm{m}$}
\QQ{$x=6.0\,\mathrm{m}$ から $x=1.0\,\mathrm{m}$ まで直線上を進んだ。}{移動距離 $5.0\,\mathrm{m}$、変位 $-5.0\,\mathrm{m}$}
\QQ{$x=-2.0\,\mathrm{m}$ から $x=3.0\,\mathrm{m}$ へ進み、さらに $x=1.0\,\mathrm{m}$ へ戻った。}{移動距離 $7.0\,\mathrm{m}$、変位 $+3.0\,\mathrm{m}$}
\QQ{$x=4.0\,\mathrm{m}$ から $x=-1.0\,\mathrm{m}$ へ進み、さらに $x=2.0\,\mathrm{m}$ へ戻った。}{移動距離 $8.0\,\mathrm{m}$、変位 $-2.0\,\mathrm{m}$}
\QQ{$x=0\,\mathrm{m}$ から $x=5.0\,\mathrm{m}$ へ進み、原点に戻った。}{移動距離 $10\,\mathrm{m}$、変位 $0\,\mathrm{m}$}

\TypeQuestionSingle{平均の速さを求めなさい。}{average_speed}
\QQRow{$18\,\mathrm{m}$ を $4.5\,\mathrm{s}$ で進む。}{$4.0\,\mathrm{m/s}$}{$72\,\mathrm{m}$ を $12\,\mathrm{s}$ で進む。}{$6.0\,\mathrm{m/s}$}
\QQRow{$270\,\mathrm{m}$ を $45\,\mathrm{s}$ で進む。}{$6.0\,\mathrm{m/s}$}{$2.4\,\mathrm{km}$ を $8.0\,\mathrm{min}$ で進む。}{$5.0\,\mathrm{m/s}$}
\QQRow{$75\,\mathrm{cm}$ を $0.50\,\mathrm{s}$ で進む。}{$1.5\,\mathrm{m/s}$}{$1.08\,\mathrm{km}$ を $3.0\,\mathrm{min}$ で進む。}{$6.0\,\mathrm{m/s}$}

\TypeQuestionSingle{平均の速度を求めなさい。ただし、右向きを正とする。}{average_velocity}
\QQRow{$x=2.0\,\mathrm{m}$ から $x=14\,\mathrm{m}$ へ $3.0\,\mathrm{s}$ で移動}{$+4.0\,\mathrm{m/s}$}{$x=10\,\mathrm{m}$ から $x=-2.0\,\mathrm{m}$ へ $4.0\,\mathrm{s}$ で移動}{$-3.0\,\mathrm{m/s}$}
\QQRow{$x=-5.0\,\mathrm{m}$ から $x=7.0\,\mathrm{m}$ へ $6.0\,\mathrm{s}$ で移動}{$+2.0\,\mathrm{m/s}$}{$x=3.0\,\mathrm{m}$ から $x=8.4\,\mathrm{m}$ へ $1.8\,\mathrm{s}$ で移動}{$+3.0\,\mathrm{m/s}$}
\QQRow{$x=-1.0\,\mathrm{m}$ から $x=-7.0\,\mathrm{m}$ へ $2.5\,\mathrm{s}$ で移動}{$-2.4\,\mathrm{m/s}$}{$x=9.0\,\mathrm{m}$ から $x=0\,\mathrm{m}$ へ $3.0\,\mathrm{s}$ で移動}{$-3.0\,\mathrm{m/s}$}

\TypeQuestionSingle{往復を含む運動について、平均の速さと平均の速度を求めなさい。}{round_trip}
\QQ{$x=0\,\mathrm{m}$ から $x=8.0\,\mathrm{m}$ へ進み、$x=3.0\,\mathrm{m}$ へ戻るのに $5.0\,\mathrm{s}$ かかった。}{平均の速さ $2.6\,\mathrm{m/s}$、平均の速度 $+0.60\,\mathrm{m/s}$}
\QQ{$x=2.0\,\mathrm{m}$ から $x=-4.0\,\mathrm{m}$ へ進み、$x=2.0\,\mathrm{m}$ へ戻るのに $6.0\,\mathrm{s}$ かかった。}{平均の速さ $2.0\,\mathrm{m/s}$、平均の速度 $0\,\mathrm{m/s}$}
\QQ{$x=-3.0\,\mathrm{m}$ から $x=1.0\,\mathrm{m}$ へ進み、$x=-5.0\,\mathrm{m}$ へ移動するのに $4.0\,\mathrm{s}$ かかった。}{平均の速さ $2.5\,\mathrm{m/s}$、平均の速度 $-0.50\,\mathrm{m/s}$}
\QQ{$x=5.0\,\mathrm{m}$ から $x=1.0\,\mathrm{m}$ へ進み、さらに $x=-2.0\,\mathrm{m}$ へ移動するのに $3.5\,\mathrm{s}$ かかった。}{平均の速さ $2.0\,\mathrm{m/s}$、平均の速度 $-2.0\,\mathrm{m/s}$}

\TypeQuestionSingle{速さの単位を変換しなさい。}{unit_conversion}
\QQRow{$36\,\mathrm{km/h}$ を $\mathrm{m/s}$ で表す。}{$10\,\mathrm{m/s}$}{$54\,\mathrm{km/h}$ を $\mathrm{m/s}$ で表す。}{$15\,\mathrm{m/s}$}
\QQRow{$90\,\mathrm{km/h}$ を $\mathrm{m/s}$ で表す。}{$25\,\mathrm{m/s}$}{$12\,\mathrm{m/s}$ を $\mathrm{km/h}$ で表す。}{$43.2\,\mathrm{km/h}$}
\QQRow{$7.5\,\mathrm{m/s}$ を $\mathrm{km/h}$ で表す。}{$27\,\mathrm{km/h}$}{$0.80\,\mathrm{m/s}$ を $\mathrm{km/h}$ で表す。}{$2.88\,\mathrm{km/h}$}

\TypeQuestionSingle{速度と時間から変位や位置を求めなさい。ただし、右向きを正とする。}{velocity_to_position}
\QQRow{速度 $+3.0\,\mathrm{m/s}$ で $4.0\,\mathrm{s}$ 進むときの変位}{$+12\,\mathrm{m}$}{速度 $-2.5\,\mathrm{m/s}$ で $6.0\,\mathrm{s}$ 進むときの変位}{$-15\,\mathrm{m}$}
\QQRow{$x=2.0\,\mathrm{m}$ から速度 $+4.0\,\mathrm{m/s}$ で $3.0\,\mathrm{s}$ 進んだ後の位置}{$x=+14\,\mathrm{m}$}{$x=-5.0\,\mathrm{m}$ から速度 $+1.5\,\mathrm{m/s}$ で $8.0\,\mathrm{s}$ 進んだ後の位置}{$x=+7.0\,\mathrm{m}$}
\QQRow{$x=10\,\mathrm{m}$ から速度 $-3.0\,\mathrm{m/s}$ で $5.0\,\mathrm{s}$ 進んだ後の位置}{$x=-5.0\,\mathrm{m}$}{$x=-6.0\,\mathrm{m}$ から速度 $-0.50\,\mathrm{m/s}$ で $8.0\,\mathrm{s}$ 進んだ後の位置}{$x=-10\,\mathrm{m}$}

\TypeQuestionSingle{速さと速度の違いに注意して、次の問いに答えなさい。}{speed_velocity_difference}
\QQ{物体が進んだ道のりを時間で割った量は、平均の速さと平均の速度のどちらか。}{平均の速さ}
\QQ{物体の変位を時間で割った量は、平均の速さと平均の速度のどちらか。}{平均の速度}
\QQ{右向きに $2.0\,\mathrm{m/s}$ で進む物体の速度を、右向きを正として表しなさい。}{$+2.0\,\mathrm{m/s}$}
\QQ{左向きに $2.0\,\mathrm{m/s}$ で進む物体の速度を、右向きを正として表しなさい。}{$-2.0\,\mathrm{m/s}$}
\QQ{物体が出発点に戻ったとき、その運動全体の変位は何になるか。}{$0$}

\TypeQuestionSingle{次の文章を読み、変位または平均の速度を求めなさい。}{application}
\QQ{東向きを正として、駅から東へ $480\,\mathrm{m}$、西へ $120\,\mathrm{m}$ 移動した。駅からの変位を求めなさい。}{$+360\,\mathrm{m}$}
\QQ{右向きを正として、$x=-12\,\mathrm{m}$ の位置から出発し、$9.0\,\mathrm{s}$ 後に $x=6.0\,\mathrm{m}$ に着いた。平均の速度を求めなさい。}{$+2.0\,\mathrm{m/s}$}
\QQ{右向きを正として、平均の速度が $-1.5\,\mathrm{m/s}$ の物体が $8.0\,\mathrm{s}$ 運動した。変位を求めなさい。}{$-12\,\mathrm{m}$}
\QQ{右向きを正として、$x=4.0\,\mathrm{m}$ の位置から速度 $-2.0\,\mathrm{m/s}$ で $5.0\,\mathrm{s}$ 進んだ。到着位置を求めなさい。}{$x=-6.0\,\mathrm{m}$}
\QQ{自転車が $0.72\,\mathrm{km}$ を $2.0\,\mathrm{min}$ で進んだ。平均の速さを $\mathrm{m/s}$ で求めなさい。}{$6.0\,\mathrm{m/s}$}
