\MajorQuestionLesson{指数法則と指数表示}
\SetRowSpace{3.5mm}

\TypeQuestionSingle{指数法則を用いて、次の式を $10^n$ の形で表しなさい。}{shisuu_housoku}
\QQRow{$10^4\times10^{-7}$}{$10^{-3}$}{$10^{-2}\div10^4$}{$10^{-6}$}
\QQRow{$\left(10^3\right)^{-2}$}{$10^{-6}$}{$\left(10^{-2}\right)^3$}{$10^{-6}$}
\QQRow{$\dfrac{1}{10^{-5}}$}{$10^5$}{$\dfrac{10^{-3}}{10^{-8}}$}{$10^5$}
\QQRow{$10^6\times10^{-2}\div10^3$}{$10^1$}{$\dfrac{10^4\times10^{-7}}{10^{-5}}$}{$10^2$}
\QQRow{$\left(10^2\times10^{-5}\right)^2$}{$10^{-6}$}{$\dfrac{\left(10^{-3}\right)^2}{10^{-8}}$}{$10^2$}

\TypeQuestionSingle{次の数や式を、$a\times10^n$（$1\leqq a<10$、$n$ は整数）の形に直しなさい。}{seikika}
%答えのaに小数第３位まであるようなものも2つほど混ぜて下さい。
\QQRow{$0.00045$}{$4.5\times10^{-4}$}{$305000$}{$3.05\times10^5$}
\QQRow{$0.031$}{$3.1\times10^{-2}$}{$600$}{$6\times10^2$}
\QQRow{$0.00406$}{$4.06\times10^{-3}$}{$900000$}{$9\times10^5$}
\QQRow{$12\times10^{-6}$}{$1.2\times10^{-5}$}{$0.35\times10^8$}{$3.5\times10^7$}
\QQRow{$728\times10^2$}{$7.28\times10^4$}{$0.0064\times10^{-3}$}{$6.4\times10^{-6}$}
\QQRow{$1000\times10^{-7}$}{$1\times10^{-4}$}{$0.0012\times10^9$}{$1.2\times10^6$}

\TypeQuestionSingle{次の指数表示された数を、指数を使わない数で表しなさい。}{shisuu_modosi}
\QQRow{$4.5\times10^{-4}$}{$0.00045$}{$7.28\times10^6$}{$7280000$}
\QQRow{$3.1\times10^{-2}$}{$0.031$}{$6.0\times10^2$}{$600$}
\QQRow{$8.09\times10^{-7}$}{$0.000000809$}{$4.2\times10^8$}{$420000000$}

\TypeQuestionSingle{次の計算をし、答えを $a\times10^n$（$1\leqq a<10$、$n$ は整数）の形で表しなさい。}{shisuu_joujo}
%答えのaに小数が7割程度混じるように(小数第２位くらいまで、整数・第一位までのと混じる感じ)。整数も合っていいけど、答えのaはかなり小数多め、っていうバランス
\QQRow{$\left(2.5\times10^3\right)\times\left(4.0\times10^{-2}\right)$}{$1\times10^2$}{$\dfrac{7.2\times10^{-4}}{2.4\times10^2}$}{$3\times10^{-6}$}
\QQRow{$\dfrac{\left(6.0\times10^8\right)\times\left(3.0\times10^{-5}\right)}{9.0\times10^2}$}{$2\times10^1$}{$\dfrac{4.8\times10^{-3}}{\left(1.2\times10^2\right)\times\left(2.0\times10^{-4}\right)}$}{$2\times10^{-1}$}
\QQRow{$\left(3.0\times10^{-2}\right)^2$}{$9\times10^{-4}$}{$\dfrac{1}{2.5\times10^{-4}}$}{$4\times10^3$}
\QQRow{$\dfrac{\left(1.5\times10^4\right)^2}{3.0\times10^6}$}{$7.5\times10^1$}{$\left(2.0\times10^{-3}\right)^3$}{$8\times10^{-9}$}

\TypeQuestionSingle{指数をそろえてから計算し、答えを $a\times10^n$（$1\leqq a<10$、$n$ は整数）の形で表しなさい。}{shisuu_kagen}
\QQRow{$3.2\times10^5+4.5\times10^4$}{$3.65\times10^5$}{$7.0\times10^{-3}-2.5\times10^{-4}$}{$6.75\times10^{-3}$}
\QQRow{$6.4\times10^6+8.0\times10^5$}{$7.2\times10^6$}{$1.25\times10^{-5}+7.5\times10^{-6}$}{$2\times10^{-5}$}
\QQRow{$4.8\times10^{-7}+3.2\times10^{-7}$}{$8\times10^{-7}$}{$8.1\times10^8-6.0\times10^7$}{$7.5\times10^8$}

\TypeQuestionSingle{各式に指数表示された値を代入し、指定された未知変数の値を指数表示で求めなさい。}{koushiki_dainyuu}
%答えのaに小数が6割程度混じるように(小数第２位くらいまで、整数・第一位までのと混じる感じ)。整数も合っていいけど、答えのaはかなり小数多め、っていうバランス。ただ小数第２位はshisuu_joujoよりは少なめで。
\QQ{$v=\dfrac{x}{t}$ に、$x=1.5\times10^3$、$t=3.0\times10^{-2}$ を代入し、$v$ を求める。}{$v=5\times10^4$}
\QQ{$p=p_0+\rho gh$ に、$p=5.0\times10^5$、$p_0=1.0\times10^5$、$g=1.0\times10^1$、$h=2.0\times10^1$ を代入し、$\rho$ を求める。}{$\rho=2\times10^3$}
\QQ{$E=\dfrac{1}{2}mv^2$ に、$E=9.0\times10^4$、$v=3.0\times10^2$ を代入し、$m$ を求める。}{$m=2\times10^0$}
\QQ{$Q=mc\Delta T$ に、$Q=8.4\times10^5$、$m=2.0\times10^2$、$c=4.2\times10^3$ を代入し、$\Delta T$ を求める。}{$\Delta T=1\times10^0$}
\QQTall{$\dfrac{1}{2}mv_0^2+mgh_0=\dfrac{1}{2}mv^2+mgh$ に、$m=1.25\times10^4$、$v_0=6.0\times10^0$、$g=1.0\times10^1$、$h_0=4.2\times10^0$、$h=1.0\times10^0$ を代入し、$v$ を求める。}{$v=1\times10^1$}
\QQ{$mgh=\mu mgx$ に、$m=2.5\times10^3$、$g=9.8\times10^0$、$h=1.2\times10^0$、$\mu=3.0\times10^{-1}$ を代入し、$x$ を求める。}{$x=4\times10^0$}
